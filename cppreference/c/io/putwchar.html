putwchar

Défini dans l'en-tête <wchar.h>

wint_t putwchar( wchar_t ch );  // (depuis C95)

Écrit un caractère large ch dans stdout.

Paramètres

  • ch — caractère large à écrire

Valeur de retour

ch en cas de succès, WEOF en cas d'échec.

Exemple

#include <locale.h>
#include <stdio.h>
#include <stdlib.h>
#include <wchar.h>

int main()
{
    setlocale(LC_ALL, "en_US.utf8");

    const wchar_t data[] =
    {
        L'\u2200', // Nom Unicode : "FOR ALL"
        L'∀',
        L'\n',
    };

    for (size_t t = 0; t != (sizeof data / sizeof(wchar_t)); ++t)
    {
        if (putwchar(data[t]) == WEOF)
        {
            puts("I/O error in putwchar");
            return EXIT_FAILURE;
        }
    }

    return EXIT_SUCCESS;
}
