Remarque : en Unicode, le bloc de caractères ASCII est appelé U+0000..U+007F Basic Latin.

Exemple

#include <stdio.h>

int main(void)
{
    puts("Printable ASCII:");
    for (int i = 32; i < 127; ++i) {
        putchar(i);
        putchar(i % 16 == 15 ? '\n' : ' ');
    }
}
