std::ranges::constant_range

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <ranges>

template< class T >
concept constant_range =
    ranges::input_range<T> &&
    /*constant-iterator*/<ranges::iterator_t<T>>;  // (1) (depuis C++23)

Concepts auxiliaires

template< class T >
concept /*constant-iterator*/ =
    std::input_iterator<T> &&
    std::same_as<std::iter_const_reference_t<T>, std::iter_reference_t<T>>;  // (2) ( exposition uniquement* )

  1. Le concept constant_range est un raffinement de range pour lequel ranges::begin renvoie un itérateur constant.

  2. Le concept /*constant-iterator*/<T> est satisfait lorsque le résultat de l'opération d'indirection de l'itérateur d'entrée est son type de référence const, ce qui implique une lecture seule.

Exemple

#include <ranges>
#include <span>
#include <string_view>
#include <vector>

// mécanismes permettant de garantir que le paramètre est un intervalle constant
// 1) un ensemble de surcharges où la version modifiable délègue à la version constante
template<std::ranges::constant_range R>
void takes_any_range1(R&& r)
{
    // R est définitivement un intervalle constant
}

template<std::ranges::range R>
void takes_any_range1(R&& r)
{
    takes_any_range1(std::views::as_const(std::forward<R>(r)));
}

// 2) un modèle de fonction unique qui masque son paramètre
template<std::ranges::range R>
void takes_any_range2(R&& _r)
{
    auto r = std::views::as_const(std::forward<R>(_r));

    // r est définitivement un intervalle constant
    // ne plus jamais utiliser _r
}

// 3) un modèle de fonction unique qui s'invoque récursivement
template<std::ranges::range R>
void takes_any_range3(R&& r)
{
    if constexpr (std::ranges::constant_range<R>)
    {
        // R est définitivement un intervalle constant
        // placer l'implémentation ici
    }
    else
        takes_any_range3(std::views::as_const(std::forward<R>(r)));
}

static_assert
(
        std::ranges::constant_range<const std::vector<int>> and
    not std::ranges::constant_range<std::vector<int>> and
        std::ranges::constant_range<std::string_view> and
    not std::ranges::constant_range<std::span<int>> and
        std::ranges::constant_range<std::span<const int>> and
    not std::ranges::constant_range<const std::span<int>>
);
