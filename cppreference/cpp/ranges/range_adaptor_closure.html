std::ranges::range_adaptor_closure

Défini dans l'en-tête <ranges>

template< class D >
    requires std::is_object_v<D> && std::same_as<D, std::remove_cv_t<D>>
class range_adaptor_closure {};  // (depuis C++23)

std::ranges::range_adaptor_closure est un modèle de classe auxiliaire permettant de définir un RangeAdaptorClosureObject.

Soit t l'objet de type T ; l'implémentation garantit que t est un objet de fermeture d'adaptateur d'intervalle si toutes les exigences sont satisfaites :

  • t est un objet fonction unaire qui prend un argument range.
  • T a exactement une classe de base publique ranges::range_adaptor_closure<T>, et T n'a aucune classe de base de type ranges::range_adaptor_closure<U> pour tout autre type U.
  • T ne satisfait pas range.

Exemple

#include <ranges>
#include <string_view>

// Définir Slice comme objet de fermeture d'adaptateur d'intervalle
struct Slice : std::ranges::range_adaptor_closure<Slice>
{
    std::size_t start = 0;
    std::size_t end = std::string_view::npos;

    constexpr std::string_view operator()(std::string_view sv) const
    {
        return sv.substr(start, end - start);
    }
};

int main()
{
    constexpr std::string_view str = "01234567";

    constexpr Slice slicer{.start = 1, .end = 6};

    // utiliser slicer comme objet fonction ordinaire
    constexpr auto sv1 = slicer(str);
    static_assert(sv1 == "12345");

    // utiliser slicer comme objet de fermeture d'adaptateur d'intervalle
    constexpr auto sv2 = str | slicer;
    static_assert(sv2 == "12345");

    // les objets de fermeture d'adaptateurs d'intervalles peuvent être composés
    constexpr auto slice_and_drop = slicer | std::views::drop(2);
    static_assert(std::string_view(str | slice_and_drop) == "345");
}
