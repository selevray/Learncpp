std::ranges::random_access_range

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <ranges>

template< class T >
concept random_access_range =
    ranges::bidirectional_range<T> && std::random_access_iterator<ranges::iterator_t<T>>;  // (depuis C++20)

Le concept random_access_range est un raffinement de range pour lequel ranges::begin renvoie un modèle de random_access_iterator.

Exemple

#include <array>
#include <deque>
#include <list>
#include <ranges>
#include <set>
#include <valarray>
#include <vector>

template<typename T> concept RAR = std::ranges::random_access_range<T>;

int main()
{
    int a[4];
    static_assert(
            RAR<std::vector<int>> and
            RAR<std::vector<bool>> and
            RAR<std::deque<int>> and
            RAR<std::valarray<int>> and
            RAR<decltype(a)> and
        not RAR<std::list<int>> and
        not RAR<std::set<int>> and
            RAR<std::array<std::list<int>,42>>
    );
}
