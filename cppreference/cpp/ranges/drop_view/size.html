std::ranges::drop_view<V>::size

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

constexpr auto size() requires ranges::sized_range<V>;  // (1) (depuis C++20)

constexpr auto size() const requires ranges::sized_range<const V>;  // (2) (depuis C++20)

Renvoie le nombre d'éléments.

Soit base_ la vue sous-jacente, count_ le compteur stocké (le nombre passé au constructeur, ou 0 si *this est construit par défaut). Équivalent à :

const auto s = ranges::size(base_);
const auto c = static_cast<decltype(s)>(count_);
return s < c ? 0 : s - c;

Paramètres

(aucune)

Valeur de retour

Le nombre d'éléments.

Exemple

#include <array>
#include <ranges>

int main()
{
    constexpr std::array a{42, 43, 44};
    static_assert(std::ranges::drop_view{std::views::all(a), 0}.size() == 3);
    static_assert(std::ranges::drop_view{std::views::all(a), 1}.size() == 2);
    static_assert(std::ranges::drop_view{std::views::all(a), 2}.size() == 1);
    static_assert(std::ranges::drop_view{std::views::all(a), 3}.size() == 0);
    static_assert(std::ranges::drop_view{std::views::all(a), 4}.size() == 0);
}
