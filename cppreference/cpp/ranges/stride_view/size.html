std::ranges::stride_view<V>::size

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

constexpr auto size() requires ranges::sized_range<V>;  // (depuis C++23)

constexpr auto size() const requires ranges::sized_range<const V>;  // (depuis C++23)

Renvoie le nombre d'éléments.

Soient base_ la vue sous-jacente et stride_ la valeur de pas stockée. Équivalent à :

return /*to-unsigned-like*/(/*div-ceil*/(ranges::distance(base_), stride_));

Paramètres

(aucune)

Valeur de retour

Le nombre d'éléments. La valeur renvoyée est calculée comme si par l'expression

(ranges::size(base_) / stride_) + ((ranges::size(base_) % stride_ ? 1 : 0).

Exemple

#include <forward_list>
#include <ranges>

int main()
{
    namespace vs = std::views;
    constexpr static auto v = {1, 2, 3, 4, 5};
    static_assert
    (
        vs::stride(v, 1).size() == 5 and
        vs::stride(v, 2).size() == 3 and
        vs::stride(v, 3).size() == 2 and
        vs::stride(v, 4).size() == 2 and
        vs::stride(v, 5).size() == 1 and
        vs::stride(v, 6).size() == 1
    );

    std::forward_list list{v};
//  auto s = vs::stride(list, 2).size(); // Erreur : n'est pas un sized_range
}
