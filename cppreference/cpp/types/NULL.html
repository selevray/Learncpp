#define NULL /* implementation-defined */

La macro NULL est une constante de pointeur nul définie par l'implémentation.

Implémentation possible

#define NULL 0
// depuis C++11
#define NULL nullptr

Notes

En C, la macro NULL peut avoir le type void*, mais cela n'est pas autorisé en C++ car les constantes de pointeur nul ne peuvent pas avoir ce type.

Exemple

#include <cstddef>
#include <iostream>
#include <type_traits>
#include <typeinfo>

class S;

int main()
{
    int* p = NULL;
    int* p2 = static_cast<std::nullptr_t>(NULL);
    void(*f)(int) = NULL;
    int S::*mp = NULL;
    void(S::*mfp)(int) = NULL;
    auto nullvar = NULL; // peut déclencher un avertissement lors de la compilation avec gcc/clang

    std::cout << "The type of nullvar is " << typeid(nullvar).name() << '\n';

    if constexpr(std::is_same_v<decltype(NULL), std::nullptr_t>)
        std::cout << "NULL implemented with type std::nullptr_t\n";
    else
        std::cout << "NULL implemented using an integral type\n";

    [](...){}(p, p2, f, mp, mfp); // < supprime les avertissements « variable inutilisée »
}
