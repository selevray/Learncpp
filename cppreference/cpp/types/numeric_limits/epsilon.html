std::numeric_limits<T>::epsilon

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

static T epsilon() throw();  // (jusqu'à C++11)

static constexpr T epsilon() noexcept;  // (depuis C++11)

Renvoie l'epsilon machine, c'est-à-dire la différence entre 1.0 et la valeur suivante représentable par le type à virgule flottante T. N'a de sens que si std::numeric_limits<T>::is_integer == false.

Exemple

Démontre l'utilisation de l'epsilon machine pour comparer des valeurs à virgule flottante pour l'égalité :

#include <algorithm>
#include <cmath>
#include <cstddef>
#include <iomanip>
#include <iostream>
#include <limits>
#include <type_traits>

template <class T>
std::enable_if_t<not std::numeric_limits<T>::is_integer, bool>
equal_within_ulps(T x, T y, std::size_t n)
{
    // Puisque `epsilon()` est la taille de l'écart (ULP, unité de la dernière place)
    // des nombres à virgule flottante dans l'intervalle [1, 2), on peut la mettre à l'échelle pour obtenir
    // la taille de l'écart dans l'intervalle [2^e, 2^{e+1}), où `e` est l'exposant
    // de `x` et `y`.

    // Si `x` et `y` ont des tailles d'écart différentes (ce qui signifie qu'ils ont
    // des exposants différents), on prend la plus petite. Prendre la plus grande
    // est également raisonnable, je suppose.
    const T m = std::min(std::fabs(x), std::fabs(y));

    // Les nombres subnormaux ont un exposant fixe, qui vaut `min_exponent - 1`.
    const int exp = m < std::numeric_limits<T>::min()
                  ? std::numeric_limits<T>::min_exponent - 1
                  : std::ilogb(m);

    // Nous considérons `x` et `y` égaux si la différence entre eux est
    // inférieure à `n` ULP.
    return std::fabs(x - y) <= n * std::ldexp(std::numeric_limits<T>::epsilon(), exp);
}

int main()
{
    double x = 0.3;
    double y = 0.1 + 0.2;
    std::cout << std::hexfloat;
    std::cout << "x = " << x << '\n';
    std::cout << "y = " << y << '\n';
    std::cout << (x == y ? "x == y" : "x != y") << '\n';
    for (std::size_t n = 0; n <= 10; ++n)
        if (equal_within_ulps(x, y, n))
        {
            std::cout << "x equals y within " << n << " ulps" << '\n';
            break;
        }
}

Output:

x = 0x1.3333333333333p-2
y = 0x1.3333333333334p-2
x != y
x equals y within 1 ulps
