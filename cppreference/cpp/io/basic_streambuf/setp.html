std::basic_streambuf<CharT,Traits>::setp

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

protected:
void setp( char_type* pbeg, char_type* pend );

Définit les valeurs des pointeurs délimitant la zone d'écriture.

Après l'appel, pbase() == pbeg, pptr() == pbeg et epptr() == pend valent tous true.

Si l'un de [pbeg,pend) n'est pas un intervalle valide, le comportement est indéfini.

Paramètres

  • pbeg — pointeur vers le nouveau début de la zone d'écriture
  • pend — pointeur vers la nouvelle fin de la zone d'écriture

Exemple

#include <array>
#include <cstddef>
#include <iostream>

// Tampon pour std::ostream implémenté par std::array
template<std::size_t size, class CharT = char>
struct ArrayedStreamBuffer : std::basic_streambuf<CharT>
{
    using Base = std::basic_streambuf<CharT>;
    using char_type = typename Base::char_type;

    ArrayedStreamBuffer()
    {
        // pointeurs de la zone d'écriture à utiliser avec « buffer »
        Base::setp(buffer.data(), buffer.data() + size);
    }

    void print_buffer()
    {
        for (char_type i : buffer)
        {
            if (i == 0)
                std::cout << "\\0";
            else
                std::cout << i;
            std::cout << ' ';
        }
        std::cout << '\n';
    }

private:
    std::array<char_type, size> buffer{}; // initialiser « buffer » par valeur
};

int main()
{
    ArrayedStreamBuffer<10> streambuf;
    std::ostream stream(&streambuf);

    stream << "hello";
    stream << ",";

    streambuf.print_buffer();
}

Output:

h e l l o , \0 \0 \0 \0
