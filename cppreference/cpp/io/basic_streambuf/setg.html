std::basic_streambuf<CharT,Traits>::setg

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

protected:
void setg( char_type* gbeg, char_type* gcurr, char_type* gend );

Définit les valeurs des pointeurs délimitant la zone de lecture.

Après l'appel, eback() == gbeg, gptr() == gcurr et egptr() == gend valent tous true.

Si l'un de [gbeg,gend), [gbeg,gcurr) et [gcurr,gend) n'est pas un intervalle valide, le comportement est indéfini.

Paramètres

  • gbeg — pointeur vers le nouveau début de la zone de lecture
  • gcurr — pointeur vers le nouveau caractère courant (pointeur de lecture) dans la zone de lecture
  • gend — pointeur vers la nouvelle fin de la zone de lecture

Exemple

#include <iostream>
#include <sstream>

class null_filter_buf : public std::streambuf
{
    std::streambuf* src;
    char ch; // tampon d'un seul octet
protected:
    int underflow()
    {
        traits_type::int_type i;
        while ((i = src->sbumpc()) == '\0')
            ; // ignorer les zéros
        if (!traits_type::eq_int_type(i, traits_type::eof()))
        {
            ch = traits_type::to_char_type(i);
            setg(&ch, &ch, &ch+1); // rendre une position de lecture disponible
        }
        return i;
    }
public:
    null_filter_buf(std::streambuf* buf) : src(buf)
    {
        setg(&ch, &ch + 1, &ch + 1); // le tampon est initialement plein
    }
};

void filtered_read(std::istream& in)
{
    std::streambuf* orig = in.rdbuf();
    null_filter_buf buf(orig);
    in.rdbuf(&buf);
    for (char c; in.get(c);)
        std::cout << c;
    in.rdbuf(orig);
}

int main()
{
    char a[] = "This i\0s \0an e\0\0\0xample";
    std::istringstream in(std::string(std::begin(a), std::end(a)));
    filtered_read(in);
}
