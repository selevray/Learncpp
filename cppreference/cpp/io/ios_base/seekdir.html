Spécifie le type de direction de déplacement dans un fichier. Les constantes suivantes sont définies :

Constante Explication

  • beg — le début d'un flux
  • end — la fin d'un flux
  • cur — la position courante de l'indicateur de position du flux

Exemple

#include <iostream>
#include <sstream>
#include <string>

int main()
{
    std::istringstream in("Hello, World!");
    std::string word1, word2, word3, word4, word5;

    in >> word1;
    in.seekg(0, std::ios_base::beg); // <- rewind
    in >> word2;
    in.seekg(1, std::ios_base::cur); // -> déplacement depuis la position courante vers la fin
    in >> word3;
    in.seekg(-6, std::ios_base::cur); // <- déplacement depuis la position courante (fin) vers le début
    in >> word4;
    in.seekg(-6, std::ios_base::end); // <- déplacement depuis la fin vers le début
    in >> word5;

    std::cout << "word1 = " << word1 << '\n'
              << "word2 = " << word2 << '\n'
              << "word3 = " << word3 << '\n'
              << "word4 = " << word4 << '\n'
              << "word5 = " << word5 << '\n';
}

Output:

word1 = Hello,
word2 = Hello,
word3 = World!
word4 = World!
word5 = World!
