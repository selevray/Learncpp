Spécialise l'algorithme std::swap pour std::unordered_map. Échange le contenu de lhs et rhs. Appelle lhs.swap(rhs).

Paramètres

  • lhs, rhs — conteneurs dont le contenu doit être échangé

Valeur de retour

(aucune)

Complexité

Constante.

Exceptions

noexcept specification:noexcept(noexcept(lhs.swap(rhs)))

(depuis C++17)

Exemple

#include <algorithm>
#include <iostream>
#include <unordered_map>

int main()
{
    std::unordered_map<int, char> alice{{1, 'a'}, {2, 'b'}, {3, 'c'}};
    std::unordered_map<int, char> bob{{7, 'Z'}, {8, 'Y'}, {9, 'X'}, {10, 'W'}};

    auto print = [](std::pair<const int, char>& n)
    {
        std::cout << ' ' << n.first << ':' << n.second;
    };

    // Affiche l'état avant swap
    std::cout << "Alice:";
    std::for_each(alice.begin(), alice.end(), print);
    std::cout << "\nBobby:";
    std::for_each(bob.begin(), bob.end(), print);
    std::cout << '\n';

    std::cout << "-- SWAP\n";
    std::swap(alice, bob);

    // Affiche l'état après swap
    std::cout << "Alice:";
    std::for_each(alice.begin(), alice.end(), print);
    std::cout << "\nBobby:";
    std::for_each(bob.begin(), bob.end(), print);
    std::cout << '\n';
}
