Retourne un itérateur vers le premier élément du unordered_map.

Si le unordered_map est vide, l'itérateur retourné sera égal à end().

range-begin-end.svg

Paramètres

(aucune)

Valeur de retour

Itérateur vers le premier élément.

Complexité

Constante.

Exemple

#include <cmath>
#include <iostream>
#include <unordered_map>

struct Node { double x, y; };

int main()
{
    Node nodes[3] = {{1, 0}, {2, 0}, {3, 0}};

    // mag est une map associant l'adresse d'un Node à sa norme dans le plan
    std::unordered_map<Node*, double> mag =
    {
        { nodes + 0, 1 },
        { nodes + 1, 2 },
        { nodes + 2, 3 }
    };

    // Remplace chaque coordonnée y de 0 par la norme
    for (auto iter = mag.begin(); iter != mag.end(); ++iter)
    {
        auto cur = iter->first; // pointeur vers Node
        cur->y = mag[cur]; // on aurait aussi pu utiliser cur->y = iter->second;
    }

    // Met à jour et affiche la norme de chaque nœud
    for (auto iter = mag.begin(); iter != mag.end(); ++iter)
    {
        auto cur = iter->first;
        mag[cur] = std::hypot(cur->x, cur->y);
        std::cout << "The magnitude of (" << cur->x << ", " << cur->y << ") is ";
        std::cout << iter->second << '\n';
    }

    // Répète ce qui précède avec la boucle for basée sur un intervalle
    for (auto i : mag)
    {
        auto cur = i.first;
        cur->y = i.second;
        mag[cur] = std::hypot(cur->x, cur->y);
        std::cout << "The magnitude of (" << cur->x << ", " << cur->y << ") is ";
        std::cout << mag[cur] << '\n';
        // Notez qu'à la différence de std::cout << iter->second << '\n'; ci-dessus,
        // std::cout << i.second << '\n'; n'affichera PAS la norme mise à jour
    }
}
