guides de déduction pour std::span

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <span>

template< class It, class EndOrSize >
span( It, EndOrSize ) -> span<std::remove_reference_t<std::iter_reference_t<It>>>;  // (depuis C++20) (jusqu'à C++26)

template< class It, class EndOrSize >
span( It, EndOrSize ) -> span<std::remove_reference_t<std::iter_reference_t<It>>,
                              /*maybe-static-ext*/<EndOrSize>>;  // (depuis C++26)

template< class T, std::size_t N >
span( T (&)[N] ) -> span<T, N>;  // (2) (depuis C++20)

template< class T, std::size_t N >
span( std::array<T, N>& ) -> span<T, N>;  // (3) (depuis C++20)

template< class T, std::size_t N >
span( const std::array<T, N>& ) -> span<const T, N>;  // (4) (depuis C++20)

template< class R >
span( R&& ) -> span<std::remove_reference_t<std::ranges::range_reference_t<R>>>;  // (5) (depuis C++20)

Les guides de déduction suivants sont fournis pour span.

  1. Permet de déduire le type d'élément à partir de la paire itérateur-sentinelle. Il permet également de déduire l'extension statique si EndOrSize satisfait integral-constant-like. (depuis C++26). Cette surcharge ne participe à la résolution de surcharge que si It satisfait contiguous_iterator.

2-4) Permet de déduire l'extension statique à partir des tableaux intégrés et de std::array.

  1. Permet de déduire le type d'élément à partir des intervalles. Cette surcharge ne participe à la résolution de surcharge que si R satisfait contiguous_range.

Exemple

#include <array>
#include <cstddef>
#include <iomanip>
#include <iostream>
#include <span>
#include <string_view>
#include <vector>

void print(std::string_view rem = "", std::size_t size_of = 0, std::size_t extent = 0)
{
    if (rem.empty())
    {
        std::cout << "name │ sizeof │ extent\n"
                     "─────┼────────┼────────\n";
        return;
    }
    std::cout << std::setw(4) << rem << " │ " << std::setw(6) << size_of << " │ ";
    if (extent == std::dynamic_extent)
        std::cout << "dynamic";
    else
        std::cout << extent;
    std::cout << '\n';
}

int main()
{
    int a[]{1, 2, 3, 4, 5};

    print();
    std::span s1{std::begin(a), std::end(a)}; // guide (1)
    print("s1", sizeof s1, s1.extent);

    std::span s2{std::begin(a), 3}; // guide (1)
    print("s2", sizeof s2, s2.extent);

#if __cplusplus > 202302L
    std::span s3{std::begin(a), std::integral_constant<std::size_t, 2>{}}; // guide (1)
    print("s3", sizeof s3, s3.extent);
#endif // C++26

    std::span s4{a}; // guide (2)
    print("s4", sizeof s4, s4.extent);

    std::span<int> s5{a}; // n'utilise pas de guide, crée un span dynamique
    print("s5", sizeof s5, s5.extent);

    std::array arr{6, 7, 8};
    std::span s6{arr}; // guide (3)
    print("s6", sizeof s6, s6.extent);
    s6[0] = 42; // OK, element_type vaut 'int'

    const std::array arr2{9, 10, 11};
    std::span s7{arr2}; // guide (4)
    print("s7", sizeof s7, s7.extent);
    // s7[0] = 42; // Error: element_type vaut 'const int'

    std::vector v{66, 69, 99};
    std::span s8{v}; // guide (5)
    print("s8", sizeof s8, s8.extent);
}
