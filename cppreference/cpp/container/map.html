    class T,

    class Allocator = std::allocator<std::pair<const Key, T>>

namespace pmr {
    template<
        class Key,
        class T,
        class Compare = std::less<Key>
    > using map = std::map<Key, T, Compare,
                           std::pmr::polymorphic_allocator<std::pair<const Key, T>>>;
}  // (2) (depuis C++17)

std::map est un conteneur associatif trié qui contient des paires clé-valeur à clés uniques. Les clés sont triées à l'aide de la fonction de comparaison Compare. Les opérations de recherche, de suppression et d'insertion ont une complexité logarithmique. Les maps sont généralement implémentées sous forme d'arbres rouge-noir.

Les itérateurs de std::map parcourent les clés par ordre croissant, où l'ordre croissant est défini par la comparaison utilisée à la construction. C'est-à-dire que, étant donnés

  • m, a std::map
  • it_l et it_r, des itérateurs déréférençables vers m, avec it_l < it_r.

m.value_comp()(*it_l, *it_r) == true (du plus petit au plus grand si l'on utilise la comparaison par défaut).

Partout où la bibliothèque standard utilise les exigences Compare, l'unicité est déterminée à l'aide de la relation d'équivalence. En termes imprécis, deux objets a et b sont considérés comme équivalents (non uniques) si aucun des deux n'est strictement inférieur à l'autre : !comp(a, b) && !comp(b, a).

std::map satisfait les exigences de Container, AllocatorAwareContainer, AssociativeContainer et ReversibleContainer.

Exemple

#include <iostream>
#include <map>
#include <string>
#include <string_view>

void print_map(std::string_view comment, const std::map<std::string, int>& m)
{
    std::cout << comment;
    // Itérer avec les fonctionnalités de C++17
    for (const auto& [key, value] : m)
        std::cout << '[' << key << "] = " << value << "; ";

// C++11 alternative:
//  for (const auto& n : m)
//      std::cout << n.first << " = " << n.second << "; ";
//
// C++98 alternative:
//  for (std::map<std::string, int>::const_iterator it = m.begin(); it != m.end(); ++it)
//      std::cout << it->first << " = " << it->second << "; ";

    std::cout << '\n';
}

int main()
{
    // Créer une map de trois paires (string, int)
    std::map<std::string, int> m{{"CPU", 10}, {"GPU", 15}, {"RAM", 20}};

    print_map("1) Initial map: ", m);

    m["CPU"] = 25; // mettre à jour une valeur existante
    m["SSD"] = 30; // insérer une nouvelle valeur
    print_map("2) Updated map: ", m);

    // Utiliser operator[] avec une clé inexistante effectue toujours une insertion
    std::cout << "3) m[UPS] = " << m["UPS"] << '\n';
    print_map("4) Updated map: ", m);

    m.erase("GPU");
    print_map("5) After erase: ", m);

    std::erase_if(m, [](const auto& pair){ return pair.second > 25; });
    print_map("6) After erase: ", m);
    std::cout << "7) m.size() = " << m.size() << '\n';

    m.clear();
    std::cout << std::boolalpha << "8) Map is empty: " << m.empty() << '\n';
}

Output:

1) Initial map: [CPU] = 10; [GPU] = 15; [RAM] = 20;
2) Updated map: [CPU] = 25; [GPU] = 15; [RAM] = 20; [SSD] = 30;
3) m[UPS] = 0
4) Updated map: [CPU] = 25; [GPU] = 15; [RAM] = 20; [SSD] = 30; [UPS] = 0;
5) After erase: [CPU] = 25; [RAM] = 20; [SSD] = 30; [UPS] = 0;
6) After erase: [CPU] = 25; [RAM] = 20; [UPS] = 0;
7) m.size() = 3
8) Map is empty: true
