std::full_extent, std::full_extent_t

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <mdspan>

struct full_extent_t { explicit full_extent_t() = default; };  // (1) (depuis C++26)

inline constexpr std::full_extent_t full_extent {};  // (2) (depuis C++26)

  1. La classe std::full_extent_t est un type de spécificateur de tranche utilisable dans std::submdspan.

  2. L'instance correspondante de std::full_extent de (1) est un spécificateur de tranche indiquant la plage complète des indices dans l'extent spécifié dans std::submdspan.

Exemple

#include <mdspan>
#include <print>

void print(auto view)
{
    static_assert(view.rank() <= 2);

    if constexpr (view.rank() == 2)
    {
        for (std::size_t i = 0; i != view.extent(0); ++i)
        {
            for (std::size_t j = 0; j != view.extent(1); ++j)
                std::print("{} ", view[i, j]);
            std::println();
        }
    }
    else if constexpr (view.rank() == 1)
    {
        for (std::size_t i = 0; i != view.extent(0); ++i)
            std::print("{} ", view[i]);
        std::println();
    }
    else
        std::println("{}", view[]);

    std::println();
}

int main()
{
    const char letters []{'A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I'};
    const std::mdspan view(letters, 3, 3);

    print(view);
    print(std::submdspan(view, std::full_extent, std::full_extent));
    print(std::submdspan(view, std::full_extent, 1));
    print(std::submdspan(view, 1, std::full_extent));
    print(std::submdspan(view, 2, 1));
}
