template< class T, class Container, class Compare, class Alloc >
struct uses_allocator<std::priority_queue<T, Compare, Container>, Alloc>
    : std::uses_allocator<Container, Alloc>::type {};  // (depuis C++11)

Fournit une spécialisation transparente du trait de type std::uses_allocator pour std::priority_queue : l'adaptateur de conteneur utilise l'allocateur si et seulement si le conteneur sous-jacent l'utilise.
