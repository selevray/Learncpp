Spécialise l'algorithme std::swap pour std::unordered_multiset. Échange le contenu de lhs et rhs. Appelle lhs.swap(rhs).

Paramètres

  • lhs, rhs — conteneurs dont le contenu doit être échangé

Valeur de retour

(aucune)

Complexité

Constante.

Exceptions

noexcept specification:noexcept(noexcept(lhs.swap(rhs)))

(depuis C++17)

Exemple

#include <algorithm>
#include <iostream>
#include <unordered_set>

int main()
{
    std::unordered_multiset<int> alice{1, 2, 3};
    std::unordered_multiset<int> bob{7, 8, 9, 10};

    auto print = [](const int& n) { std::cout << ' ' << n; };

    // Affiche l'état avant swap
    std::cout << "Alice:";
    std::for_each(alice.begin(), alice.end(), print);
    std::cout << "\nBobby:";
    std::for_each(bob.begin(), bob.end(), print);
    std::cout << '\n';

    std::cout << "-- SWAP\n";
    std::swap(alice, bob);

    // Affiche l'état après swap
    std::cout << "Alice:";
    std::for_each(alice.begin(), alice.end(), print);
    std::cout << "\nBobby:";
    std::for_each(bob.begin(), bob.end(), print);
    std::cout << '\n';
}
