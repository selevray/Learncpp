### Classes {#Classes}

  • locale — ensemble de facettes polymorphes encapsulant les différences culturelles (classe)

  • wstring_convert (C++11) (deprecated in C++17) (removed in C++26) — effectue des conversions entre une chaîne de caractères larges et une chaîne d'octets (modèle de classe)
  • wbuffer_convert (C++11) (deprecated in C++17) (removed in C++26) — effectue une conversion entre un tampon de flux d'octets et un tampon de flux de caractères larges (modèle de classe)

Classe std::locale

class locale
{
public:
    // types:
    class facet;
    class id;
    typedef int category;
    static const category // les valeurs affectées ici sont données à titre d'exposition uniquement
        none     = 0,
        collate  = 0x010,
        ctype    = 0x020,
        monetary = 0x040,
        numeric  = 0x080,
        time     = 0x100,
        messages = 0x200,
        all = collate | ctype | monetary | numeric | time | messages;

    // construction/copie/destruction :
    locale() noexcept;
    locale(const locale& other) noexcept;
    explicit locale(const char* std_name);
    explicit locale(const string& std_name);
    locale(const locale& other, const char* std_name, category);
    locale(const locale& other, const string& std_name, category);
    template <class Facet> locale(const locale& other, Facet* f);
    locale(const locale& other, const locale& one, category);
    ~locale();

    // non virtuelle
    const locale& operator=(const locale& other) noexcept;
    template <class Facet> locale combine(const locale& other) const;

    // opérations sur les locales :
    basic_string<char>
    name() const;
    bool operator==(const locale& other) const;
    bool operator!=(const locale& other) const;
    template <class CharT, class Traits, class Allocator>
    bool operator()(const basic_string<CharT,Traits,Allocator>& s1,
                    const basic_string<CharT,Traits,Allocator>& s2) const;

    // objets locale globaux :
    static
    locale global(const locale&);
    static const locale& classic();
};
