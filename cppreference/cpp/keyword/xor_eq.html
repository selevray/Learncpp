C++ keyword: xor_eq

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Utilisation

  • opérateurs alternatifs : en remplacement de ^=

Exemple

#include <bitset>
#include <iostream>

using bin = std::bitset<8>;

void show(bin z, const char* s, int n)
{
    if (n == 0) std::cout << "┌────────────┬──────────┐\n";
    if (n <= 2) std::cout << "│ " <<s<<  " │ " <<z<<" │\n";
    if (n == 2) std::cout << "└────────────┴──────────┘\n";
}

int main()
{
    bin x{"01011010"}; show(x, "x         ", 0);
    bin y{"00111100"}; show(y, "y         ", 1);
    x xor_eq y;        show(x, "x xor_eq y", 2);
}

Output:

┌────────────┬──────────┐
│ x          │ 01011010 │
│ y          │ 00111100 │
│ x xor_eq y │ 01100110 │
└────────────┴──────────┘
