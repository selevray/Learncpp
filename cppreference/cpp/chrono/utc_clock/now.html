std::chrono::utc_clock::now

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

static std::chrono::time_point<std::chrono::utc_clock> now();  // (depuis C++20)

Renvoie un point temporel représentant l'instant courant. Le résultat est calculé comme par std::chrono::utc_clock::from_sys(std::chrono::system_clock::now()). Les implémentations peuvent utiliser une valeur plus précise de l'heure UTC.

Valeur de retour

Un point temporel représentant l'heure actuelle.

Exemple

#include <chrono>
#include <cstddef>
#include <iomanip>
#include <iostream>
#include <numeric>
#include <vector>

volatile int sink; // empêche l'optimisation

void do_some_work(std::size_t size)
{
    std::vector<int> v(size, 42);
    sink = std::accumulate(v.begin(), v.end(), 0); // garantit l'effet de bord
}

int main()
{
    std::cout << std::fixed << std::setprecision(9) << std::left;
    for (auto size{1ull}; size < 1000'000'000ull; size *= 100)
    {
        const auto start = std::chrono::utc_clock::now();
        do_some_work(size);
        const auto end = std::chrono::utc_clock::now();
        const std::chrono::duration<double> diff = end - start;
        std::cout << "start = " << start << "; end = " << end << ";\n";
        std::cout << "diff = " << diff << "; size = " << size << '\n';
    }
}
