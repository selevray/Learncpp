constexpr bool operator==( const std::chrono::weekday_last& x,
                           const std::chrono::weekday_last& y ) noexcept;  // (depuis C++20)

Compare les deux valeurs weekday_last x et y.

L'opérateur != est synthétisé à partir de operator==.

Valeur de retour

x.weekday() == y.weekday()

Exemple

#include <chrono>
#include <iostream>

int main()
{
    constexpr auto wdl1{std::chrono::Tuesday[std::chrono::last]};
    constexpr std::chrono::weekday_last wdl2{std::chrono::weekday(2)};
    std::cout << std::boolalpha
              << (wdl1 == wdl2) << ' '
              << (wdl1 == std::chrono::Wednesday[std::chrono::last]) << '\n';
}
