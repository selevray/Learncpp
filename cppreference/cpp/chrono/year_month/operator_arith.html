constexpr std::chrono::year_month&
    operator-=( const std::chrono::years& dy ) const noexcept;  // (3) (depuis C++20)

constexpr std::chrono::year_month&
    operator-=( const std::chrono::months& dm ) const noexcept;  // (4) (depuis C++20)

Modifie le point temporel que *this représente, de la durée dy ou dm.

  1. Équivalent à *this = *this + dy;.

  2. Équivalent à *this = *this + dm;.

  3. Équivalent à *this = *this - dy;.

  4. Équivalent à *this = *this - dm;.

Pour les durées convertibles à la fois en std::chrono::years et en std::chrono::months, les surcharges years (1,3) sont préférées si l'appel serait sinon ambigu.

Exemple

#include <cassert>
#include <chrono>

int main()
{
    auto ym{std::chrono::day(1)/7/2023};

    ym -= std::chrono::years{2};
    assert(ym.month() == std::chrono::July);
    assert(ym.year() == std::chrono::year(2021));

    ym += std::chrono::months{7};
    assert(ym.month() == std::chrono::month(2));
    assert(ym.year() == std::chrono::year(2022));
}
