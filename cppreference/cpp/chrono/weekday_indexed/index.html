std::chrono::weekday_indexed::index

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

constexpr unsigned index() const noexcept;  // (depuis C++20)

Accède à l'indice stocké dans *this.

Valeur de retour

L'indice stocké dans *this.

Exemple

#include <chrono>
#include <iostream>

int main()
{
    std::cout << std::boolalpha;

    std::chrono::weekday_indexed wdi {std::chrono::Tuesday[2]}; // 2e mardi d'un mois
    std::cout << (std::chrono::year_month_day{wdi/10/2019} ==
                  std::chrono::year_month_day{std::chrono::October/8/2019}) << ' ';

    wdi = {wdi.weekday()[wdi.index() + 2]}; // 2nd Tuesday => 4th Tuesday
    std::cout << (std::chrono::year_month_day{wdi/10/2019} ==
                  std::chrono::year_month_day{std::chrono::October/22/2019}) << '\n';
}
