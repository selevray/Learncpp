Récupère les valeurs de champs stockées dans cet objet year_month_weekday_last.

Valeur de retour

  1. Renvoie la valeur std::chrono::year stockée.

  2. Renvoie la valeur std::chrono::month stockée.

  3. Renvoie la valeur std::chrono::weekday stockée.

  4. std::chrono::weekday_last(weekday())

Exemple

#include <chrono>
#include <iostream>
using namespace std::chrono;

int main()
{
    auto ymwdl{Tuesday[last]/November/2022};
    auto wdl = ymwdl.weekday_last();
    wdl = (wdl.weekday() + days(1))[last];
    ymwdl = {ymwdl.year() + years(1), ymwdl.month() - months(2), wdl};
    std::cout << year_month_day(ymwdl) << '\n';
}
