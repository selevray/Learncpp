template< class CharT, class Traits >
std::basic_ostream<CharT, Traits>&
    operator<<( std::basic_ostream<CharT, Traits>& os,
                const std::chrono::year_month_weekday_last& ymwdl );  // (depuis C++20)

Écrit une représentation textuelle de ymwdl dans os, comme si par

os << std::format(os.getloc(), STATICALLY_WIDEN<CharT>("{}/{:L}/{:L}"), ymwdl.year(), ymwdl.month(), ymwdl.weekday_last());

où STATICALLY_WIDEN<CharT>("{}/{:L}/{:L}") vaut "{}/{:L}/{:L}" si CharT est char, et L"..." si CharT est wchar_t.

os
