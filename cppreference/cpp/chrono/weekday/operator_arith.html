constexpr std::chrono::weekday& operator+=( const std::chrono::days& d ) noexcept;  // (1) (depuis C++20)

constexpr std::chrono::weekday& operator-=( const std::chrono::days& d ) noexcept;  // (2) (depuis C++20)

Ajoute ou soustrait d.count() à la valeur du jour de la semaine, en réduisant le résultat modulo 7 à un entier dans l'intervalle [0,6].

  1. Effectue *this = *this + d;.

  2. Effectue *this = *this - d;.

Valeur de retour

Une référence à ce weekday après modification.

Notes

Après un appel à l'une de ces fonctions, ok() vaut toujours true si aucun dépassement n'a eu lieu pendant l'opération.

Exemple

#include <chrono>
#include <iostream>

int main()
{
    std::cout << std::boolalpha;

    std::chrono::weekday wd{5}; // Friday vaut 5
    wd += std::chrono::days(2);
    std::cout << (wd == std::chrono::weekday(0)) << ' '
              << (wd == std::chrono::Sunday) << ' ';

    wd -= std::chrono::days(3);
    std::cout << (wd == std::chrono::weekday(4)) << ' '
              << (wd == std::chrono::Thursday) << ' ';

    wd = std::chrono::Wednesday;
    wd += std::chrono::days{8}; // ((3 += 8 == 11) % 7) == 4;
    std::cout << (wd == std::chrono::Thursday) << ' ';

    wd -= (std::chrono::Sunday - std::chrono::Tuesday); // -= -2
    // (4 -= -2) == 6
    std::cout << (wd == std::chrono::Saturday) << '\n';
}

true true true true true true
