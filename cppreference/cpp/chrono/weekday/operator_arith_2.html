                                          const std::chrono::days& d ) noexcept;  // (1) (depuis C++20)

constexpr std::chrono::weekday operator+( const std::chrono::days& d,
                                          const std::chrono::weekday& wd ) noexcept;  // (2) (depuis C++20)

constexpr std::chrono::weekday operator-( const std::chrono::weekday& wd,
                                          const std::chrono::days& d ) noexcept;  // (3) (depuis C++20)

constexpr std::chrono::days operator-( const std::chrono::weekday& wd1,
                                       const std::chrono::weekday& wd2 ) noexcept;  // (4) (depuis C++20)

1,2) Ajoute d.count() jours à wd. La valeur du jour de la semaine contenue dans le résultat est calculée en évaluant d'abord static_cast<long long>(wd.c_encoding()) + d.count() puis en la réduisant modulo 7 à un entier dans l'intervalle [0,6].

  1. Soustrait d.count() jours de wd. Équivalent à return wd + -d;.

  2. Si wd1.ok() et wd2.ok() valent tous deux true, renvoie une valeur std::chrono::days d telle que d.count() soit dans l'intervalle [0,6] et wd2 + d == wd1. Sinon la valeur renvoyée est non spécifiée.

Valeur de retour

1-3) Un std::chrono::weekday contenant une valeur de jour de la semaine calculée comme décrit ci-dessus.

  1. Un std::chrono::days représentant la distance entre wd1 et wd2.

Notes

Tant que le calcul ne déborde pas, (1-3) renvoie toujours un weekday valide, même si wd.ok() vaut false.

Exemple

#include <chrono>
#include <iostream>

int main()
{
    std::cout << std::boolalpha;

    std::chrono::weekday wd{4};
    wd = wd + std::chrono::days(2);
    std::cout << (wd == std::chrono::weekday(6)) << ' '
              << (wd == std::chrono::Saturday) << ' ';

    wd = wd - std::chrono::days(3);
    std::cout << (wd == std::chrono::weekday(3)) << ' '
              << (wd == std::chrono::Wednesday) << ' ';

    wd = std::chrono::Tuesday;
    wd = wd + std::chrono::days{8}; // (((2 + 8) == 10) % 7) == 3;
    std::cout << (wd == std::chrono::Wednesday) << ' ';

    wd = wd + (std::chrono::Sunday - std::chrono::Thursday); // (3 + 3) == 6
    std::cout << (wd == std::chrono::Saturday) << '\n';
}
