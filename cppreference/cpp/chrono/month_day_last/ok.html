Vérifie si l'objet month stocké dans *this est valide.

Valeur de retour

month().ok()

Exemple

#include <cassert>
#include <chrono>

int main()
{
    auto mdl{std::chrono::February/std::chrono::last};
    assert(mdl.ok());
    mdl = {std::chrono::month(42)/std::chrono::last};
    assert(!mdl.ok());
}
