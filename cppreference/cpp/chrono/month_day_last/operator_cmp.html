constexpr bool operator==( const std::chrono::month_day_last& x,
                           const std::chrono::month_day_last& y ) noexcept;  // (1) (depuis C++20)

constexpr std::strong_ordering operator<=>( const std::chrono::month_day_last& x,
                                            const std::chrono::month_day_last& y ) noexcept;  // (2) (depuis C++20)

Compare les deux valeurs month_day_last x et y.

Les opérateurs <, <=, >, >= et != sont synthétisés à partir de operator<=> et operator== respectivement.

Valeur de retour

  1. x.month() == y.month()

  2. x.month() <=> y.month()

Exemple

#include <chrono>
#include <iostream>

int main()
{
    constexpr auto mdl1{std::chrono::February/std::chrono::last};
    constexpr std::chrono::month_day_last mdl2{std::chrono::month(2)};
    std::cout << std::boolalpha << (mdl1 == mdl2) << '\n';

    static_assert(mdl1 <= mdl2);
}
