constexpr std::chrono::month month() const noexcept;  // (depuis C++20)

Récupère une copie de l'objet std::chrono::month stocké dans *this.

Valeur de retour

Une copie de l'objet std::chrono::month stocké dans *this.

Exemple

#include <chrono>
#include <iostream>

using namespace std::chrono;

int main()
{
    std::cout << std::boolalpha;

    auto mdl{February/last}; // Dernier jour de février
    auto ymdl{year(2020) / mdl};
    std::cout << (year_month_day{ymdl} == year_month_day{February/29/2020}) << ' ';

    mdl = (mdl.month() + months(1)) / last; // Dernier jour du mois suivant, en 2020
    ymdl = year(2020) / mdl;
    std::cout << (year_month_day{ymdl} == year_month_day{March/31/2020}) << '\n';
}
