constexpr std::chrono::month month() const noexcept;  // (1) (depuis C++20)

constexpr std::chrono::day day() const noexcept;  // (1) (depuis C++20)

Récupère les valeurs du mois et du jour stockées dans cet objet month_day.

Valeur de retour

  1. Renvoie la valeur std::chrono::month stockée.

  2. Renvoie la valeur std::chrono::day stockée.

Exemple

#include <chrono>

int main()
{
    constexpr auto md{std::chrono::July/15};
    static_assert(md.month() == std::chrono::month(7));
    static_assert(md.day() == std::chrono::day(15));
}
