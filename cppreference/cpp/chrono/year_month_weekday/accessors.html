Récupère les valeurs de champs stockées dans cet objet year_month_weekday.

Valeur de retour

  1. Renvoie la valeur std::chrono::year stockée.

  2. Renvoie la valeur std::chrono::month stockée.

  3. Renvoie la valeur std::chrono::weekday stockée.

  4. Renvoie l'indice de jour de la semaine stocké.

  5. weekday()[index()]

Exemple

#include <cassert>
#include <chrono>

int main()
{
    constexpr auto ym{std::chrono::year(2021)/std::chrono::January};
    constexpr auto wdi{std::chrono::Wednesday[1]};
    auto ymwdi{ym/wdi};
    const auto index{ymwdi.index() + 1};
    auto weekday{ymwdi.weekday() + std::chrono::days(1)};
    ymwdi = {ymwdi.year()/ymwdi.month()/weekday[index]};
    // Deuxième jeudi de janvier 2021
    assert(std::chrono::year_month_day{ymwdi} == std::chrono::January/14/2021);
}
