Compare les deux std::chrono::year x et y.

Les opérateurs <, <=, >, >= et != sont synthétisés à partir de operator<=> et operator== respectivement.

Valeur de retour

  1. int(x) == int(y)

  2. int(x) <=> int(y)

Exemple

#include <chrono>
#include <iostream>

int main()
{
    using namespace std::chrono;

    constexpr year y1{2020};
    constexpr year y2{2021};

    std::cout << std::boolalpha << (y1 != y2) << '\n';

    static_assert((2020y < 2023y) && (2020y == 2020y) && (2020y <= 2023y) &&
                  (2023y > 2020y) && (2023y != 2020y) && (2023y >= 2020y));
}
