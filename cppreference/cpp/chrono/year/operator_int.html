constexpr explicit operator int() const noexcept;  // (depuis C++20)

Retourne la valeur d'année stockée dans *this.

Valeur de retour

La valeur d'année stockée dans *this.

Exemple

#include <chrono>
#include <iostream>
using namespace std::chrono;

int main()
{
    constexpr std::chrono::year y{2020};
    std::cout << "The year is: " << static_cast<int>(y) << '\n';

    const year_month_day ymd{floor<days>(system_clock::now())};
    const std::chrono::year this_year{ymd.year()};
    std::cout << "This year is: " << int(this_year) << '\n';
}
