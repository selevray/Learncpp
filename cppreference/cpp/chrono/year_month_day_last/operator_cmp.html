                           const std::chrono::year_month_day_last& y ) noexcept;  // (1) (depuis C++20)

constexpr std::strong_ordering
    operator<=>( const std::chrono::year_month_day_last& x,
                 const std::chrono::year_month_day_last& y ) noexcept;  // (2) (depuis C++20)

Compare les deux valeurs year_month_day_last x et y. Il s'agit d'une comparaison lexicographique : la year() est comparée en premier, puis month().

Les opérateurs <, <=, >, >= et != sont synthétisés à partir de operator<=> et operator== respectivement.

Valeur de retour

  1. x.year() == y.year() && x.month() == y.month()

  2. x.year() <=> y.year() != 0 ? x.year() <=> y.year() : x.month() <=> y.month()

Notes

Si x et y représentent tous deux des dates valides (x.ok() && y.ok() == true), le résultat de la comparaison lexicographique est cohérent avec l'ordre du calendrier.

Exemple

#include <cassert>
#include <chrono>
#include <iostream>

int main()
{
    auto ymdl1{11/std::chrono::last/2020};
    auto mdl{std::chrono::last/std::chrono::November};
    auto ymdl2{mdl/2020};
    assert(ymdl1 == ymdl2);

    ymdl1 -= std::chrono::months{2};
    ymdl2 -= std::chrono::months{1};
    assert(ymdl1 < ymdl2);
}
