std::hash<std::chrono::time_point>

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <chrono>

template< class Clock, class Duration >
struct hash<std::chrono::time_point<Clock, Duration>>;  // (depuis C++26)

La spécialisation de modèle de std::hash pour std::chrono::time_point permet aux utilisateurs d'obtenir les empreintes (hashes) des objets de type std::chrono::time_point<Clock, Duration>. Cette spécialisation est activée si et seulement si std::hash<Duration> est activé.

Notes

Macro de test de fonctionnalité Valeur Std Fonctionnalité
__cpp_lib_chrono 202306L (C++26) Prise en charge du hachage pour les classes de valeurs std::chrono

Exemple

#include <chrono>
#include <cstddef>
#include <iostream>
#include <string>
#include <thread>
#include <unordered_map>

struct my_system_clock : std::chrono::system_clock
{
    using time_point = std::chrono::time_point<my_system_clock>;

    static time_point now() noexcept
    {
        return time_point{std::chrono::system_clock::now().time_since_epoch()};
    }

    template<class CharT, class Traits>
    friend auto operator<<(std::basic_ostream<CharT, Traits>& os, const time_point& tp)
        -> decltype(os)
    {
        return os << std::chrono::system_clock::time_point{tp.time_since_epoch()};
    }
};

using my_system_clock_tp = std::chrono::time_point<my_system_clock>;

#if __cpp_lib_chrono < 202306L
// une spécialisation personnalisée de std::hash peut être injectée dans l'espace de noms std
template<>
struct std::hash<my_system_clock_tp>
{
    std::size_t operator()(const my_system_clock_tp& d) const noexcept
    {
        return d.time_since_epoch().count();
    }
};
#endif

int main()
{
    using namespace std::chrono_literals;

    std::unordered_map<my_system_clock_tp, std::string> log;

    for (int i{}; i != 4; ++i)
    {
        std::this_thread::sleep_for(100ms);
        log[my_system_clock::now()] = "event #" + std::to_string(i);
    }

    for (auto const& [time, message] : log)
        std::cout << '[' << time << "], message: " << message << '\n';
}
