std::filesystem::directory_entry::path

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

const std::filesystem::path& path() const noexcept;  // (depuis C++17)

operator const std::filesystem::path& () const noexcept;  // (depuis C++17)

Renvoie le chemin complet vers lequel pointe l'entrée de répertoire.

Paramètres

(aucune)

Valeur de retour

Le chemin complet vers lequel pointe l'entrée de répertoire.

Exemple

#include <filesystem>
#include <fstream>
#include <iostream>

namespace fs = std::filesystem;

std::string get_stem(const fs::path& p) { return p.stem().string(); }
void create_file(const fs::path& p) { std::ofstream o{p}; }

int main()
{
    const fs::path dir{"tmp_dir"};
    fs::create_directory(dir);
    create_file(dir / "one");
    create_file(dir / "two");
    create_file(dir / "three");

    for (const auto& file : fs::directory_iterator(dir))
    {
        // Conversion explicite
        std::cout << get_stem(file.path()) << '\n';

        // Conversion implicite
        std::cout << get_stem(file) << '\n';
    }

    fs::remove_all(dir);
}
