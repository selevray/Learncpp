Produit des valeurs booléennes aléatoires, selon la fonction de probabilité discrète. La probabilité de true est

Exemple

#include <iomanip>
#include <iostream>
#include <map>
#include <random>
#include <string>

int main()
{
    std::random_device rd;
    std::mt19937 gen(rd());
    // donne « true » 1 fois sur 4
    // donne « false » 3 fois sur 4
    std::bernoulli_distribution d(0.25);

    std::map<bool, int> hist;
    for (int n = 0; n < 10000; ++n)
        ++hist[d(gen)];

    std::cout << std::boolalpha;
    for (auto const& [key, value] : hist)
        std::cout << std::setw(5) << key << ' '
                  << std::string(value / 500, '*') << '\n';
}
