std::seed_seq::generate

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

template< class RandomIt >
void generate( RandomIt begin, RandomIt end );  // (depuis C++11)

Génère des graines non biaisées en remplissant l'intervalle de sortie [begin,end) avec des valeurs entières non signées de 32 bits, à partir des graines (éventuellement biaisées) stockées dans v.

  • Si begin == end est true, ne fait rien.
  • Sinon, génère les graines selon l'algorithme de génération décrit ci-dessous.

Si std::iterator_traits<RandomIt>::value_type n'est pas un type entier non signé, ou si sa largeur est inférieure à 32, le programme est mal formé.

Si RandomIt ne satisfait pas aux exigences de LegacyRandomAccessIterator, ou s'il n'est pas mutable, le comportement est indéfini.

Algorithme de génération

Étant donné les valeurs et opérations suivantes :

Valeur Définition

  • z — v ﻿.size()
  • n — end - begin
  • m — std::max(z + 1, n)
  • t — (n >= 623) ? 11 : (n >= 68) ? 7 : (n >= 39) ? 5 : (n >= 7) ? 3 : (n - 1) / 2
  • p — (n - t) / 2
  • q — p + t

Opération Définition

  • (\scriptsize \mathsf{xor} )xor — XOR bit à bit intégré XOR bit à bit
  • (\scriptsize \mathsf{rshift} )rshift — décalage à droite bit à bit intégré décalage à droite bit à bit
  • (\scriptsize T(x) )T(x) — (\scriptsize x\ \mathsf{xor}\ (x\ \mathsf{rshift}\ 27) )x xor (x rshift 27)

L'algorithme de génération comprend les étapes suivantes, où (\scriptsize S_i )Si désigne begin[i % n], (\scriptsize V_i )Vi désigne v ﻿[i] :

  1. Affecte à chaque élément de l'intervalle de sortie la valeur 0x8b8b8b8b.

  2. Pour chaque entier k dans [0,m), effectue les opérations suivantes dans l'ordre :

  3. Soit (\scriptsize r_1 )r1 égal à (\scriptsize 1664525 \cdot T(S_k\ \mathsf{xor}\ S_{k+p}\ \mathsf{xor}\ S_{k-1}) )1664525·T(Sk xor Sk+p xor Sk-1).

  4. Soit (\scriptsize r_2 )r2 égal à (\scriptsize r_1 + j )r1+j, où (\scriptsize j )j vaut :

  • (\scriptsize z )z, si (\scriptsize k=0 )k=0
  • (\scriptsize (k \mod n)+V_{k-1} )(k mod n)+Vk-1, si (\scriptsize 0<k⩽z )0<k⩽z
  • (\scriptsize k \mod n )k mod n, si (\scriptsize z<k )z<k

  1. Affecte à (\scriptsize S_{k+p} )Sk+p la valeur (\scriptsize (S_{k+p}+r_1) \mod 2^{32} )(Sk+p+r1) mod 232
    .

  2. Affecte à (\scriptsize S_{k+q} )Sk+q la valeur (\scriptsize (S_{k+q}+r_2) \mod 2^{32} )(Sk+q+r2) mod 232
    .

  3. Affecte à (\scriptsize S_k )Sk la valeur (\scriptsize r_2 \mod 2^{32} )r2 mod 232
    .

  4. Pour chaque entier k dans [m,m + n), effectue les opérations suivantes dans l'ordre :

  5. Soit (\scriptsize r_3 )r3 égal à (\scriptsize 1566083941 \cdot T(S_k+S_{k+p}+S_{k-1}) )1566083941·T(Sk+Sk+p+Sk-1).

  6. Soit (\scriptsize r_4 )r4 égal à (\scriptsize r_3-(k \mod n) )r3-(k mod n).

  7. Affecte à (\scriptsize S_{k+p} )Sk+p la valeur (\scriptsize (S_{k+p}\ \mathsf{xor}\ r_3) \mod 2^{32} )(Sk+p xor r3) mod 232
    .

  8. Affecte à (\scriptsize S_{k+q} )Sk+q la valeur (\scriptsize (S_{k+q}\ \mathsf{xor}\ r_4) \mod 2^{32} )(Sk+q xor r4) mod 232
    .

  9. Affecte à (\scriptsize S_k )Sk la valeur (\scriptsize r_4 \mod 2^{32} )r4 mod 232
    .

Paramètres

  • begin, end — les itérateurs désignant l'intervalle de sortie

Exceptions

Ne lève que les exceptions levées par les opérations RandomIt sur begin et end.

Notes

L'algorithme de génération est adapté de la séquence d'initialisation du générateur Mersenne Twister de Makoto Matsumoto et Takuji Nishimura, en intégrant les améliorations apportées par Mutsuo Saito en 2007.

Exemple

#include <algorithm>
#include <cassert>
#include <cstdint>
#include <iostream>
#include <random>

// Prototype de la partie principale de std::seed_seq...
struct seed_seq
{
    std::vector<std::uint32_t> v;

    seed_seq(std::initializer_list<std::uint32_t> const il) : v{il} {}

    template<typename RandomIt>
    void generate(RandomIt first, RandomIt last)
    {
        if (first == last)
            return;
        //
        // En supposant v = {1, 2, 3, 4, 5} et distance(first, last) == 10.
        //
        // Étape 1 : remplir avec 0x8b8b8b8b
        // seeds = {2341178251, 2341178251, 2341178251, 2341178251, 2341178251,
        //          2341178251, 2341178251, 2341178251, 2341178251, 2341178251}
        //
        std::fill(first, last, 0x8b8b8b8b);
        //
        // Étape 2 :
        // n = 10, s = 5, t = 3, p = 3, q = 6, m = 10
        //
        const std::uint32_t n = last - first;
        const std::uint32_t s = v.size();
        const std::uint32_t t = (n < 7) ? (n - 1) / 2
                              : (n < 39) ? 3
                              : (n < 68) ? 5
                              : (n < 623) ? 7
                              : 11;
        const std::uint32_t p = (n - t) / 2;
        const std::uint32_t q = p + t;
        const std::uint32_t m = std::max(s + 1, n);
        //
        // Première itération, k = 0 ; r1 = 1371501266, r2 = 1371501271
        //
        // seeds = {1371501271, 2341178251, 2341178251, 3712679517, 2341178251,
        //          2341178251, 3712679522, 2341178251, 2341178251, 2341178251}
        //
        // Itérations de k = 1 à k = 5 (r2 = r1 + k % n + v[k - 1])
        //
        // r1 = 2786190137, 3204727651, 4173325571, 1979226628, 401983366
        // r2 = 2786190139, 3204727655, 4173325577, 1979226636, 401983376
        //
        // seeds = {3350727907, 3188173515, 3204727655, 4173325577, 1979226636,
        //           401983376, 3591037797, 2811627722, 1652921976, 2219536532}
        //
        // Itérations de k = 6 à k = 9 (r2 = r1 + k % n)
        //
        // r1 = 2718637909, 1378394210, 2297813071, 1608643617
        // r2 = 2718637915, 1378394217, 2297813079, 1608643626
        //
        // seeds = { 434154821, 1191019290, 3237041891, 1256752498, 4277039715,
        //          2010627002, 2718637915, 1378394217, 2297813079, 1608643626}
        //
        auto begin_mod = [first, n](std::uint32_t u) -> decltype(*first)&
        {
            return first[u % n]; // c.-à-d. begin[x] est pris modulo n
        };
        auto T = [](std::uint32_t x) { return x ^ (x >> 27); };

        for (std::uint32_t k = 0, r1, r2; k < m; ++k)
        {
            r1 = 1664525 * T(begin_mod(k) ^ begin_mod(k + p) ^ begin_mod(k - 1));
            r2 = (k == 0) ? r1 + s
               : (k <= s) ? r1 + k % n + v[k - 1]
               :            r1 + k % n;
            begin_mod(k + p) += r1;
            begin_mod(k + q) += r2;
            begin_mod(k) = r2;
        }
        //
        // Étape 3
        // itérations de k = 10 à k = 19, utilisant ^= pour modifier la sortie
        //
        // r1 = 1615303485, 3210438310, 893477041, 2884072672, 1918321961,
        // r2 = 1615303485, 3210438309, 893477039, 2884072669, 1918321957
        //
        // seeds = { 303093272, 3210438309,  893477039, 2884072669, 1918321957,
        //          1117182731, 1772877958, 2669970405, 3182737656, 4094066935}
        //
        // r1 =  423054846, 46783064, 3904109085, 1534123446, 1495905687
        // r2 =  423054841, 46783058, 3904109078, 1534123438, 1495905678
        //
        // seeds = { 4204997637, 4246533866, 1856049002, 1129615051, 690460811,
        //           1075771511,   46783058, 3904109078, 1534123438, 1495905678}
        //
        for (std::uint32_t k = m, r3, r4; k < m + n; ++k)
        {
            r3 = 1566083941 * T(begin_mod(k) + begin_mod(k + p) + begin_mod(k - 1));
            r4 = r3 - k % n;
            begin_mod(k+p) ^= r3;
            begin_mod(k+q) ^= r4;
            begin_mod(k) = r4;
        }
    }
};

int main()
{
    const auto input = std::initializer_list<std::uint32_t>{1, 2, 3, 4, 5};
    const auto output_size = 10;

    // utilisant la version std de seed_seq
    std::seed_seq seq(input);
    std::vector<std::uint32_t> seeds(output_size);
    seq.generate(seeds.begin(), seeds.end());
    for (const std::uint32_t n : seeds)
        std::cout << n << '\n';

    // utilisant une version personnalisée de seed_seq
    seed_seq seq2(input);
    std::vector<std::uint32_t> seeds2(output_size);
    seq2.generate(seeds2.begin(), seeds2.end());

    assert(seeds == seeds2);
}
