std::chi_squared_distribution

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <random>

template< class RealType = double >
class chi_squared_distribution;  // (depuis C++11)

La chi_squared_distribution produit des nombres aléatoires (\small x>0)x>0 selon la distribution du khi-deux :

({\small f(x;n) = }\frac{x^{(n/2)-1}\exp{(-x/2)} }{\Gamma{(n/2)}2^{n/2} })f(x;n) =
x(n/2)-1 e-x/2
Γ(n/2) 2n/2

(\small\Gamma)Γ est la fonction Gamma (voir aussi std::tgamma) et (\small n)n sont les degrés de liberté (par défaut 1).

std::chi_squared_distribution satisfait toutes les exigences de RandomNumberDistribution.

Paramètres de modèle

  • RealType — Le type de résultat produit par le générateur. L'effet est indéfini si ce n'est pas l'un de float, double ou long double.

Exemple

#include <algorithm>
#include <cmath>
#include <iomanip>
#include <iostream>
#include <map>
#include <random>
#include <vector>

template<int Height = 5, int BarWidth = 1, int Padding = 1, int Offset = 0, class Seq>
void draw_vbars(Seq&& s, const bool DrawMinMax = true)
{
    static_assert(0 < Height and 0 < BarWidth and 0 <= Padding and 0 <= Offset);

    auto cout_n = [](auto&& v, int n = 1)
    {
        while (n-- > 0)
            std::cout << v;
    };

    const auto [min, max] = std::minmax_element(std::cbegin(s), std::cend(s));

    std::vector<std::div_t> qr;
    for (typedef decltype(*std::cbegin(s)) V; V e : s)
        qr.push_back(std::div(std::lerp(V(0), 8 * Height,
                                        (e - *min) / (*max - *min)), 8));

    for (auto h{Height}; h-- > 0; cout_n('\n'))
    {
        cout_n(' ', Offset);

        for (auto dv : qr)
        {
            const auto q{dv.quot}, r{dv.rem};
            unsigned char d[]{0xe2, 0x96, 0x88, 0}; // Bloc plein : '█'
            q < h ? d[0] = ' ', d[1] = 0 : q == h ? d[2] -= (7 - r) : 0;
            cout_n(d, BarWidth), cout_n(' ', Padding);
        }

        if (DrawMinMax && Height > 1)
            Height - 1 == h ? std::cout << "┬ " << *max:
                          h ? std::cout << "│ "
                            : std::cout << "┴ " << *min;
    }
}

int main()
{
    std::random_device rd{};
    std::mt19937 gen{rd()};

    auto χ2 = [&gen](const float dof)
    {
        std::chi_squared_distribution<float> d{dof /* n */};

        const int norm = 1'00'00;
        const float cutoff = 0.002f;

        std::map<int, int> hist{};
        for (int n = 0; n != norm; ++n)
            ++hist[std::round(d(gen))];

        std::vector<float> bars;
        std::vector<int> indices;
        for (auto const& [n, p] : hist)
            if (float x = p * (1.0 / norm); cutoff < x)
            {
                bars.push_back(x);
                indices.push_back(n);
            }

        std::cout << "dof = " << dof << ":\n";

        for (draw_vbars<4, 3>(bars); int n : indices)
            std::cout << std::setw(2) << n << "  ";
        std::cout << "\n\n";
    };

    for (float dof : {1.f, 2.f, 3.f, 4.f, 6.f, 9.f})
        χ2(dof);
}
