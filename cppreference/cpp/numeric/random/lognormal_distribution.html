Les paramètres m et s sont, respectivement, la moyenne et l'écart type du logarithme naturel de x.

std::lognormal_distribution satisfait toutes les exigences de RandomNumberDistribution.

Exemple

#include <cmath>
#include <iomanip>
#include <iostream>
#include <map>
#include <random>
#include <string>

int main()
{
    std::random_device rd;
    std::mt19937 gen(rd());

    std::lognormal_distribution<> d(1.6, 0.25);

    std::map<int, int> hist;
    for (int n = 0; n < 1e4; ++n)
        ++hist[std::round(d(gen))];

    for (std::cout << std::fixed << std::setprecision(1); auto [x, y] : hist)
        std::cout << std::hex << x << ' ' << std::string(y / 200, '*') << '\n';
}
