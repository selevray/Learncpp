std::wbuffer_convert<Codecvt,Elem,Tr>::~wbuffer_convert

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

~wbuffer_convert();

Détruit l'objet wbuffer_convert et supprime cvtptr ﻿.

Notes

Certaines implémentations sont capables de supprimer n'importe quelle facette, y compris les facettes spécifiques aux locales dotées de destructeurs protégés. D'autres implémentations exigent que la facette ait un destructeur public, à l'instar des facettes indépendantes de la locale de <codecvt>.

Exemple

#include <codecvt>
#include <iostream>
#include <locale>
#include <utility>

// Wrapper utilitaire pour adapter les facettes liées aux locales à la conversion wstring/wbuffer
template<class Facet>
struct deletable_facet : Facet
{
    template<class... Args>
    deletable_facet(Args&&... args) : Facet(std::forward<Args>(args)...) {}
    ~deletable_facet() {}
};

int main()
{
// Conversion GB18030 / UCS4, en utilisant directement la facette liée à la locale
//  typedef std::codecvt_byname<char32_t, char, std::mbstate_t> gbfacet_t;
// Erreur de compilation : « appel d'un destructeur protégé de codecvt_byname<> dans ~wbuffer_convert »
//  std::wbuffer_convert<gbfacet_t, char32_t>
//      gbto32(std::cout.rdbuf(), new gbfacet_t("zh_CN.gb18030"));

// Facette de conversion GB18030 / UCS4 utilisant une facette à destructeur public
    typedef deletable_facet<std::codecvt_byname<char32_t, char, std::mbstate_t>> gbfacet_t;
    std::wbuffer_convert<gbfacet_t, char32_t>
        gbto32(std::cout.rdbuf(), new gbfacet_t("zh_CN.gb18030"));
} // destructeur appelé ici
