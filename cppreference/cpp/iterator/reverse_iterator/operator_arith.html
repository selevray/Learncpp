Incrémente ou décrémente l'itérateur sous-jacent en sens inverse.

Surcharge Équivalent à
(1) --current; return *this;
(2) ++current; return *this;
(3) reverse_iterator tmp = *this; --current; return tmp;
(4) reverse_iterator tmp = *this; ++current; return tmp;
(5) return reverse_iterator(current - n);
(6) return reverse_iterator(current + n);
(7) current -= n; return *this;
(8) current += n; return *this;

Paramètres

  • n — position relative à la position courante

Valeur de retour

Comme décrit ci-dessus.

Exemple

#include <iostream>
#include <iterator>
#include <list>
#include <vector>

int main()
{
    std::vector v{0, 1, 2, 3, 4};
    auto rv = std::reverse_iterator{v.rbegin()};
    std::cout << *(++rv) << ' '; // 3
    std::cout << *(--rv) << ' '; // 4
    std::cout << *(rv + 3) << ' '; // 1
    rv += 3;
    std::cout << rv[0] << ' '; // 1
    rv -= 3;
    std::cout << rv[0] << '\n'; // 4

    std::list l{5, 6, 7, 8};
    auto rl = std::reverse_iterator{l.rbegin()};
    std::cout << *(++rl) << ' '; // OK: 3
    std::cout << *(--rl) << '\n'; // OK: 4
    // Les instructions suivantes provoquent une erreur de compilation car le
    // itérateur sous-jacent ne modélise pas l'itérateur à accès aléatoire :
//  *(rl + 3) = 13;
//  rl += 3;
//  rl -= 3;
}

Output:

3 4 1 1 4
7 8
