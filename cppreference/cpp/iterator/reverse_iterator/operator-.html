template< class Iter1, class Iter2 >
typename reverse_iterator<Iter>::difference_type
    operator-( const reverse_iterator<Iter1>& lhs,
               const reverse_iterator<Iter2>& rhs );  // (jusqu'à C++11)

template< class Iter1, class Iter2 >
auto operator-( const reverse_iterator<Iter1>& lhs,
                const reverse_iterator<Iter2>& rhs )
    -> decltype(rhs.base() - lhs.base());  // (depuis C++11) (constexpr depuis C++17)

Renvoie la distance entre deux adaptateurs d'itérateurs.

Paramètres

  • lhs, rhs — adaptateurs d'itérateurs dont calculer la différence

Valeur de retour

rhs.base() - lhs.base()

Notes

operator- renvoie rhs.base() - lhs.base() plutôt que lhs.base() - rhs.base() car il s'agit d'un itérateur inverse.

Exemple

#include <iostream>
#include <iterator>
#include <list>
#include <vector>

int main()
{
    std::vector vec{0, 1, 2, 3};
    std::reverse_iterator<std::vector<int>::iterator>
        vec_ri1{std::reverse_iterator{vec.rbegin()}},
        vec_ri2{std::reverse_iterator{vec.rend()}};
    std::cout << (vec_ri2 - vec_ri1) << ' '; // 4
    std::cout << (vec_ri1 - vec_ri2) << '\n'; // -4

    std::list lst{5, 6, 7, 8};
    std::reverse_iterator<std::list<int>::iterator>
        lst_ri1{std::reverse_iterator{lst.rbegin()}},
        lst_ri2{std::reverse_iterator{lst.rend()}};
//  auto n = (lst_ri1 - lst_ri2); // Erreur : les itérateurs sous-jacents ne
                                  //        modélise pas un itérateur à accès aléatoire
}

Output:

4 -4
