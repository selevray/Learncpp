std::experimental::apply

Défini dans l'en-tête <experimental/tuple>

template< class F, class Tuple >
constexpr decltype(auto) apply(F&& f, Tuple&& t);  // (TS des fondamentaux de la bibliothèque)

Invoque l'objet Callable f avec un tuple d'arguments.

Paramètres

  • f — Callable objet à invoquer
  • t — tuple dont les éléments sont utilisés comme arguments de f

Valeur de retour

Ce que renvoie f.

Implémentation possible

namespace detail
{
    template<class F, class Tuple, std::size_t... I>
    constexpr decltype(auto) apply_impl(F&& f, Tuple&& t, std::index_sequence<I...>)
    {
        return std::invoke(std::forward<F>(f), std::get<I>(std::forward<Tuple>(t))...);
        // Remarque : std::invoke est une fonctionnalité de C++17
    }
} // espace de noms detail

template<class F, class Tuple>
constexpr decltype(auto) apply(F&& f, Tuple&& t)
{
    return detail::apply_impl(std::forward<F>(f), std::forward<Tuple>(t),
        std::make_index_sequence<std::tuple_size_v<std::decay_t<Tuple>>>{});
}

Exemple

#include <iostream>
#include <tuple>

template<typename... Ts>
void print_tuple(const std::tuple<Ts...> &tuple)
{
    std::apply([](const auto&... elem)
    {
        ((std::cout << elem << '\n'), ...);
    }, tuple);
}

int main()
{
    const std::tuple<int, char> t = std::make_tuple(5, 'a');
    print_tuple(t);
}

Output:

5
a
