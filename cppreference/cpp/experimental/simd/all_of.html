std::experimental::all_of, std::experimental::any_of, std::experimental::none_of, std::experimental::some_of

Défini dans l'en-tête <experimental/simd>

template< class T, class Abi >
bool all_of( const simd_mask<T, Abi>& k );  // (1) (TS de parallélisme v2)

template< class T, class Abi >
bool any_of( const simd_mask<T, Abi>& k );  // (2) (TS de parallélisme v2)

template< class T, class Abi >
bool none_of( const simd_mask<T, Abi>& k );  // (3) (TS de parallélisme v2)

template< class T, class Abi >
bool some_of( const simd_mask<T, Abi>& k );  // (4) (TS de parallélisme v2)

  1. Renvoie true si toutes les valeurs de k valent true.

  2. Renvoie true si au moins une valeur de k vaut true.

  3. Renvoie true si toutes les valeurs de k valent false.

  4. Renvoie true si au moins une valeur de k vaut true mais que toutes les valeurs de k ne valent pas true.

Paramètres

  • k — le simd_mask auquel appliquer la réduction

Valeur de retour

Le résultat booléen de la réduction horizontale.

Exemple

#include <cassert>
#include <experimental/simd>

namespace stq = std::experimental;

int main()
{
    using mask = stq::fixed_size_simd_mask<int, 4>;

    mask mask1{false}; // = {0, 0, 0, 0}
    assert
    (
        stq::none_of(mask1) == true &&
        stq::any_of(mask1) == false &&
        stq::some_of(mask1) == false &&
        stq::all_of(mask1) == false
    );

    mask mask2{true}; // = {1, 1, 1, 1}
    assert
    (
        stq::none_of(mask2) == false &&
        stq::any_of(mask2) == true &&
        stq::some_of(mask2) == false &&
        stq::all_of(mask2) == true
    );

    mask mask3{true};
    mask3[0] = mask3[1] = false; // mask3 = {0, 0, 1, 1}
    assert
    (
        stq::none_of(mask3) == false &&
        stq::any_of(mask3) == true &&
        stq::some_of(mask3) == true &&
        stq::all_of(mask3) == false
    );
}
