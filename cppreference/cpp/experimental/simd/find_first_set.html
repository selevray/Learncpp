std::experimental::find_first_set, std::experimental::find_last_set

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <experimental/simd>

template< class T, class Abi >
int find_first_set( const simd_mask<T, Abi>& k );  // (1) (TS de parallélisme v2)

template< class T, class Abi >
int find_last_set( const simd_mask<T, Abi>& k );  // (2) (TS de parallélisme v2)

  1. Renvoie le plus petit indice i où k[i] vaut true.

  2. Renvoie le plus grand indice i où k[i] vaut true.

Le comportement est indéfini si any_of(k) est false.

Paramètres

  • k — le simd_mask auquel appliquer la réduction

Valeur de retour

Un int dans l'intervalle [0,simd_size_v<T, Abi>).

Exemple

#include <cstddef>
#include <experimental/simd>
#include <iostream>

namespace stdx = std::experimental;

template<typename Abi>
int find(stdx::simd_mask<Abi> const& v)
{
    if (stdx::any_of(v))
        return find_first_set(v);
    return -1;
}

int main()
{
    stdx::simd_mask<short> a{0};
    a[2] = a[a.size() - 2] = 1;

    for (std::size_t i = 0; i < a.size(); ++i)
        std::cout << a[i] << ' ';
    std::cout << '\n';

    std::cout << "find_first_set: " << stdx::find_first_set(a) << '\n';
    std::cout << "find_last_set: " << stdx::find_last_set(a) << '\n';
    std::cout << "find: " << find(a) << '\n';
    a[2] = 0;
    std::cout << "find: " << find(a) << '\n';
    a[a.size() - 2] = 0;
    std::cout << "find: " << find(a) << '\n';
}
