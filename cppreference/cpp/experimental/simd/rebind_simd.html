std::experimental::rebind_simd, std::experimental::resize_simd

template< class T, class V >
struct rebind_simd;  // (1) (TS de parallélisme v2)

template< int N, class V >
struct resize_simd;  // (2) (TS de parallélisme v2)

Crée un type simd ou simd_mask avec un type d'élément ou une taille différents. Le nouveau type utilise probablement une étiquette ABI différente de V::abi_type.

  1. Change le type d'élément en T et conserve la taille inchangée.

  2. Change la taille en N et conserve le type d'élément inchangé.

Paramètres de modèle

  • T — le nouveau type d'élément ; un type arithmétique autre que bool
  • N — le nouveau nombre d'éléments
  • V — un type simd ou simd_mask

Types membres

Nom Définition

  • type — type simd ou simd_mask avec un type d'élément différent (1) ou une taille différente (2)

Types auxiliaires

template< class T, class V >
using rebind_simd_t = typename rebind_simd<T, V>::type;  // (TS de parallélisme v2)

template< int N, class V >
using resize_simd_t = typename resize_simd<N, V>::type;  // (TS de parallélisme v2)

Exemple

#include <experimental/simd>
#include <iostream>

namespace stdx = std::experimental;
using floatv = stdx::native_simd<float>;

// utiliser une précision double en interne
floatv dp(floatv x)
{
    using doublev = stdx::rebind_simd_t<double, floatv>;
    return stdx::static_simd_cast<floatv>(stdx::simd_cast<doublev>(x) - 1.234);
}

template<class T>
stdx::resize_simd_t<T::size() / 2, T> partial_reduction(T x)
{
    auto [lo, hi] = stdx::split<stdx::resize_simd_t<T::size() / 2, T>>(x);
    return lo + hi;
}

int main()
{
    floatv x([](auto i) { return 1.234f + std::numeric_limits<float>::epsilon() * i; });
    x = dp(x);
    const auto y = partial_reduction(x);
    for (unsigned i = 0; i < y.size(); ++i)
        std::cout << y[i] << ' ';
    std::cout << '\n';
}
