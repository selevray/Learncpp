Applique l'opérateur d'affectation composée donné élément par élément à chaque élément correspondant des opérandes, de sorte que pour tous les i dans l'intervalle [0,size()) le résultat est équivalent à :

  1. lhs[i] += rhs[i]

  2. lhs[i] -= rhs[i]

  3. lhs[i] *= rhs[i]

  4. lhs[i] /= rhs[i]

  5. lhs[i] %= rhs[i]

  6. lhs[i] &= rhs[i]

  7. lhs[i] |= rhs[i]

  8. lhs[i] ^= rhs[i]

  9. lhs[i] <<= rhs[i]

  10. lhs[i] <<= n

  11. lhs[i] >>= rhs[i]

  12. lhs[i] >>= n

Paramètres

  • lhs — opérandes de gauche
  • rhs — opérandes de droite
  • n — nombre de bits de décalage de chaque élément de lhs

Valeur de retour

lhs

Exemple

Calcule cosh(β)² - sinh(β)² (== 1) pour un angle arbitraire β, sur des données SIMD.

#include <cstddef>
#include <experimental/simd>
#include <iomanip>
#include <iostream>
#include <string_view>
namespace stq = std::experimental;

void print(const std::string_view rem, const auto x)
{
    std::cout << rem << '\n' << std::fixed;
    for (std::size_t t{}; t != x.size(); ++t)
        std::cout << std::setw(10) << x[t] << ((t + 1) % 8 ? ',' : '\n');
    std::cout << '\n';
}

int main()
{
    constexpr auto N = 32UZ;
    using elem_t = long double;

    stq::fixed_size_simd<elem_t, N> x{[](elem_t i) { return i / N; }};
    stq::fixed_size_simd<elem_t, N> y{x};
    x = stq::cosh(x);
    y = stq::sinh(y);
    x *= x;
    y *= y;
    print("cosh(x)²:", x);
    print("sinh(x)²:", y);
    auto z = x - y;
    print("cosh(x)² - sinh(x)²:", z);
    std::cout << "reduce(z) = " << stq::reduce(z) << '\n';
}
