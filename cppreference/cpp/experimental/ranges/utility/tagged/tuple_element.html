template< std::size_t N, class Base, class... Tags >
struct tuple_element<N, std::experimental::ranges::tagged<Base, Tags...>>
    : std::tuple_element<N, Base> {};  // (ranges TS)

La spécialisation partielle de std::tuple_element pour tagged fournit un accès à la compilation aux types des éléments de tagged, en utilisant une syntaxe semblable à celle des tuples. Elle délègue simplement à std::tuple_element<N, Base>.
