template< Readable I, IndirectRegularUnaryInvocable<I> Proj >
struct projected {
    using value_type =
        std::remove_cv_t<std::remove_reference_t<ranges::indirect_result_of_t<Proj&(I)>>>;
    ranges::indirect_result_of_t<Proj&(I)> operator*() const;
};
template< WeaklyIncrementable I, class Proj >
struct difference_type<projected<I, Proj>> {
    using type = ranges::difference_type_t<I>;
};  // (ranges TS)

Le modèle de classe projected regroupe un type Readable I et une fonction Proj en un nouveau type Readable dont le type reference est le résultat de l'application de Proj au type reference de I. Il n'existe que pour faciliter la spécification des contraintes, et son operator*() peut donc ne pas être réellement défini.
