template< class T >
concept bool /*dereferenceable*/ = requires(T& t) { {*t} -> auto&&; };  // ( exposition uniquement* )

template< /*dereferenceable*/ T >
using reference_t = decltype(*declval<T&>());  // (1) (ranges TS)

template< /*dereferenceable*/ T >
    requires requires(T& t) { { ranges::iter_move(t) } -> auto&&; }
using rvalue_reference_t = decltype(ranges::iter_move(declval<T&>()));  // (2) (ranges TS)

template< Readable T >
using iter_common_reference_t = ranges::common_reference_t<ranges::reference_t<T>,

  1. Obtient le type référence d'un type déréférençable T.

  2. Obtient le type référence rvalue d'un type déréférençable T, c'est-à-dire le type de valeur de retour de ranges::iter_move.

  3. Calcule le type référence commun du type Readable. C'est le type référence commun de son type référence et d'une référence lvalue vers son type valeur.

Notes

La contrainte -> auto&& vérifie que le type de l'expression n'est pas void.
