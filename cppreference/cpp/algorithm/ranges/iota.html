std::ranges::iota, std::ranges::iota_result

Types auxiliaires

template< class O, class T >
using iota_result = ranges::out_value_result<O, T>;  // (3) (depuis C++23)

Remplit l'intervalle [first,last) de valeurs croissant séquentiellement, en commençant par value et en évaluant de façon répétée ++value.

Opération équivalente :

*(first)     = value;
*(first + 1) = ++value;
*(first + 2) = ++value;
*(first + 3) = ++value;
...

Paramètres

  • first, last — l'intervalle des éléments à remplir avec des valeurs croissant séquentiellement à partir de value
  • value — valeur initiale à stocker ; l'expression ++value doit être bien formée

Valeur de retour

{last, value + ranges::distance(first, last)}

Complexité

Exactement last - first incréments et affectations.

Implémentation possible

struct iota_fn
{
    template<std::input_or_output_iterator O, std::sentinel_for<O> S,
            std::weakly_incrementable T>
    requires std::indirectly_writable<O, const T&>
    constexpr iota_result<O, T> operator()(O first, S last, T value) const
    {
        while (first != last)
        {
            *first = as_const(value);
            ++first;
            ++value;
        }
        return {std::move(first), std::move(value)};
    }

    template<std::weakly_incrementable T, std::ranges::output_range<const T&> R>
    constexpr iota_result<std::ranges::borrowed_iterator_t<R>, T>
    operator()(R&& r, T value) const
    {
        return (*this)(std::ranges::begin(r), std::ranges::end(r), std::move(value));
    }
};

inline constexpr iota_fn iota;

Exemple

Utilise le vector d'itérateurs (std::vector<std::list<T>::iterator>) comme intermédiaire pour mélanger les éléments de std::list, car ranges::shuffle ne peut pas être appliqué directement à std::list.

#include <algorithm>
#include <functional>
#include <iostream>
#include <list>
#include <numeric>
#include <random>
#include <vector>

template <typename Proj = std::identity>
void println(auto comment, std::ranges::input_range auto&& range, Proj proj = {})
{
    for (std::cout << comment; auto const &element : range)
        std::cout << proj(element) << ' ';
    std::cout << '\n';
}

int main()
{
    std::list<int> list(8);

    // Remplit la liste avec des valeurs croissantes : 0, 1, 2, ..., 7
    std::ranges::iota(list, 0);
    println("List: ", list);

    // Un vector d'itérateurs (voir le commentaire de l'exemple)
    std::vector<std::list<int>::iterator> vec(list.size());

    // Remplit avec des itérateurs vers les éléments consécutifs de la liste
    std::ranges::iota(vec.begin(), vec.end(), list.begin());

    std::ranges::shuffle(vec, std::mt19937 {std::random_device {}()});
    println("List viewed via vector: ", vec, [](auto it) { return *it; });
}

Sortie possible :

List: 0 1 2 3 4 5 6 7
List viewed via vector: 5 7 6 0 1 3 4 2
