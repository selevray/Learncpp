std::swap(std::tuple)

template< class... Types >
void swap( std::tuple<Types...>& lhs,
           std::tuple<Types...>& rhs ) noexcept(/* voir ci-dessous */);  // (depuis C++11) (jusqu'à C++20)

template< class... Types >
constexpr void swap( std::tuple<Types...>& lhs,
                     std::tuple<Types...>& rhs ) noexcept(/* voir ci-dessous */);  // (depuis C++20)

template< class... Types >
constexpr void swap( const std::tuple<Types...>& lhs,
                     const std::tuple<Types...>& rhs ) noexcept(/* voir ci-dessous */);  // (2) (depuis C++23)

Échange le contenu de lhs et rhs. Équivalent à lhs.swap(rhs).

  1. Cette surcharge ne participe à la résolution de surcharge que si std::is_swappable_v<Ti> est true pour tout i de 0 à sizeof...(Types).

  2. Cette surcharge ne participe à la résolution de surcharge que si std::is_swappable_v<const Ti> est true pour tout i de 0 à sizeof...(Types).

(depuis C++17)

Paramètres

  • lhs, rhs — tuples dont le contenu doit être échangé

Valeur de retour

(aucune)

Exceptions

noexcept specification:noexcept(noexcept(lhs.swap(rhs)))

Exemple

#include <iostream>
#include <string>
#include <tuple>

int main()
{
    std::tuple<int, std::string, float> p1{42, "ABCD", 2.71}, p2;
    p2 = std::make_tuple(10, "1234", 3.14);

    auto print_p1_p2 = [&](auto rem)
    {
        std::cout << rem
                  << "p1 = {" << std::get<0>(p1)
                  << ", "     << std::get<1>(p1)
                  << ", "     << std::get<2>(p1) << "}, "
                  << "p2 = {" << std::get<0>(p2)
                  << ", "     << std::get<1>(p2)
                  << ", "     << std::get<2>(p2) << "}\n";
    };

    print_p1_p2("Before p1.swap(p2): ");
    p1.swap(p2);
    print_p1_p2("After  p1.swap(p2): ");
    swap(p1, p2);
    print_p1_p2("After swap(p1, p2): ");
}

Output:

Before p1.swap(p2): p1 = {42, ABCD, 2.71}, p2 = {10, 1234, 3.14}
After  p1.swap(p2): p1 = {10, 1234, 3.14}, p2 = {42, ABCD, 2.71}
After swap(p1, p2): p1 = {42, ABCD, 2.71}, p2 = {10, 1234, 3.14}
