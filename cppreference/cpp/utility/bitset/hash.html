std::hash<std::bitset>

Défini dans l'en-tête <bitset>

template< std::size_t N >
struct hash<std::bitset<N>>;  // (depuis C++11)

La spécialisation de modèle de std::hash pour std::bitset<N> permet aux utilisateurs d'obtenir les hachages d'objets de type std::bitset<N>.

Exemple

Le code suivant montre une sortie possible d'une fonction de hachage appliquée à plusieurs bitsets :

#include <bitset>
#include <functional>
#include <iostream>

int main()
{
    std::bitset<4> b1{0}, b2{42};
    std::bitset<8> b3{0}, b4{42};

    std::hash<std::bitset<4>> hash_fn4;
    std::hash<std::bitset<8>> hash_fn8;
    using bin64 = std::bitset<64>;

    std::cout << std::hex
              << bin64{hash_fn4(b1)} << " = " << hash_fn4(b1) << '\n'
              << bin64{hash_fn4(b2)} << " = " << hash_fn4(b2) << '\n'
              << bin64{hash_fn8(b3)} << " = " << hash_fn8(b3) << '\n'
              << bin64{hash_fn8(b4)} << " = " << hash_fn8(b4) << '\n';
}
