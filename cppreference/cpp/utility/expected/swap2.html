swap(std::expected)

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

friend constexpr void swap( expected& lhs, expected& rhs ) noexcept(/*voir ci-dessous*/);  // (depuis C++23)

Surcharge l'algorithme std::swap pour std::expected. Échange l'état de lhs avec celui de rhs. Appelle effectivement lhs.swap(rhs).

Cette surcharge ne participe à la résolution de surcharge que si lhs.swap(rhs) est valide.

Cette fonction n'est pas visible par la recherche non qualifiée ordinaire ni par la recherche qualifiée, et ne peut être trouvée que par la recherche dépendante des arguments lorsque std::expected<T, E> est une classe associée aux arguments.

Paramètres

  • lhs, rhs — objets expected dont les états doivent être échangés

Valeur de retour

(aucune)

Exceptions

noexcept specification:noexcept(noexcept(lhs.swap(rhs)))

Exemple

#include <expected>
#include <iostream>
#include <string>

using Ex = std::expected<std::string, int>;

void show(const Ex& ex1, const Ex& ex2)
{
    for (int i{}; i < 2; ++i)
    {
        std::cout << (i ? "ex2" : "ex1");
        if (const Ex& ex = (i ? ex2 : ex1); ex.has_value())
            std::cout << ".has_value() = " << *ex << '\n';
        else
            std::cout << ".error() = " << ex.error() << '\n';
    }
}

int main()
{
    Ex ex1("\N{DOG FACE}");
    Ex ex2{"\N{BONE}"};
    show(ex1, ex2);
    swap(ex1, ex2);
    std::cout << "swap(ex1, ex2);\n";
    show(ex1, ex2);
    std::cout << '\n';

    ex2 = std::unexpected(13);
    show(ex1, ex2);
    swap(ex1, ex2);
    std::cout << "swap(ex1, ex2);\n";
    show(ex1, ex2);
    std::cout << '\n';

    ex2 = std::unexpected(19937);
    show(ex1, ex2);
    swap(ex1, ex2);
    std::cout << "swap(ex1, ex2);\n";
    show(ex1, ex2);
    std::cout << '\n';
}
