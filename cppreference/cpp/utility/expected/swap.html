std::expected<T,E>::swap

Modèle primaire

constexpr void swap( expected& other ) noexcept(/* voir ci-dessous */);  // (1) (depuis C++23)

void spécialisation partielle

constexpr void swap( expected& other ) noexcept(/* voir ci-dessous */);  // (2) (depuis C++23)

Échange le contenu avec celui de other.

  1. Les valeurs contenues sont échangées comme suit :

Valeur de
has_value() | Valeur de other.has_value()

true | false

true | using std::swap;swap(val, rhs.val); | voir ci-dessous

false | other.swap(*this); | using std::swap;swap(unex, rhs.unex);

Si has_value() est true et other.has_value() est false, équivalent à :

// Cas 1 : les constructions par déplacement des valeurs inattendues ne lèvent pas d'exception : // “other.unex” sera restauré si la construction de “other.val” échoue if constexpr (std::is_nothrow_move_constructible_v<E>) { E temp(std::move(other.unex)); std::destroy_at(std::addressof(other.unex)); try { std::construct_at(std::addressof(other.val), std::move(val)); // peut lever une exception std::destroy_at(std::addressof(val)); std::construct_at(std::addressof(unex), std::move(temp)); } catch(...) { std::construct_at(std::addressof(other.unex), std::move(temp)); throw; } } // Cas 2 : les constructions par déplacement des valeurs attendues ne lèvent pas d'exception : // « this->val » sera restauré si la construction de « this->unex » échoue else { T temp(std::move(val)); std::destroy_at(std::addressof(val)); try { std::construct_at(std::addressof(unex), std::move(other.unex)); // peut lever une exception std::destroy_at(std::addressof(other.unex)); std::construct_at(std::addressof(other.val), std::move(temp)); } catch(...) { std::construct_at(std::addressof(val), std::move(temp)); throw; } } has_val = false; rhs.has_val = true;

Cette surcharge ne participe à la résolution de surcharge que si toutes les valeurs suivantes sont true :

  • std::is_swappable_v<T>
  • std::is_swappable_v<E>
  • std::is_move_constructible_v<T> && std::is_move_constructible_v<E>
  • std::is_nothrow_move_constructible_v<T> || std::is_nothrow_move_constructible_v<E>
  1. Les valeurs inattendues sont échangées comme suit :

Valeur de
has_value() | Valeur de other.has_value()

true | false

true | using std::swap;swap(val, rhs.val); | std::construct_at(std::addressof(unex), std::move(rhs.unex));std::destroy_at(std::addressof(rhs.unex));has_val = false;rhs.has_val = true;

false | other.swap(*this); | using std::swap;swap(unex, rhs.unex);

Cette surcharge ne participe à la résolution de surcharge que si std::is_swappable_v<E> et std::is_move_constructible_v<E> sont tous deux true.

Paramètres

  • other — l'objet expected avec lequel échanger le contenu

Exceptions

1)noexcept specification:noexcept( std::is_nothrow_move_constructible_v<T> && std::is_nothrow_swappable_v<T> && std::is_nothrow_move_constructible_v<E> && std::is_nothrow_swappable_v<E> )

2)noexcept specification:noexcept( std::is_nothrow_move_constructible_v<E> && std::is_nothrow_swappable_v<E> )

Exemple

#include <expected>
#include <iostream>
#include <string>

using Ex = std::expected<std::string, int>;

void show(const Ex& ex1, const Ex& ex2)
{
    for (int i{}; i < 2; ++i)
    {
        std::cout << (i ? "ex2" : "ex1");
        if (const Ex& ex = (i ? ex2 : ex1); ex.has_value())
            std::cout << ".has_value() = " << *ex << '\n';
        else
            std::cout << ".error() = " << ex.error() << '\n';
    }
}

int main()
{
    Ex ex1("\N{CAT FACE}");
    Ex ex2{"\N{GREEN HEART}"};
    show(ex1, ex2);
    ex1.swap(ex2);
    std::cout << "ex1.swap(ex2);\n";
    show(ex1, ex2);
    std::cout << '\n';

    ex2 = std::unexpected(13);
    show(ex1, ex2);
    std::cout << "ex1.swap(ex2);\n";
    ex1.swap(ex2);
    show(ex1, ex2);
    std::cout << '\n';

    ex2 = std::unexpected(19937);
    show(ex1, ex2);
    std::cout << "ex1.swap(ex2);\n";
    ex1.swap(ex2);
    show(ex1, ex2);
}
