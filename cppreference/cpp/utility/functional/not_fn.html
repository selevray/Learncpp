std::not_fn

Défini dans l'en-tête <functional>

template< class F >
/* unspecified */ not_fn( F&& f );  // (1) (depuis C++17) (constexpr depuis C++20)

template< auto ConstFn >
constexpr /* unspecified */ not_fn() noexcept;  // (2) (depuis C++26)

  1. Crée un enveloppeur d'appel transmettant qui renvoie la négation de l'objet appelable qu'il contient.

  2. Crée un enveloppeur d'appel transmettant qui renvoie la négation de la cible appelable déterminée statiquement. Le programme est mal formé si ConstFn est un pointeur nul ou un pointeur nul vers membre.

Paramètres

  • f — l'objet à partir duquel l'objet Callable contenu dans le wrapper est construit

Exigences sur les types

  • std::decay_t<F> doit satisfaire aux exigences de Callable et de MoveConstructible.
  • std::is_constructible_v<std::decay_t<F>, F> doit être true.

Valeur de retour

  1. Un objet fonction de type non spécifié T. Il possède les membres suivants.

std::not_fn type de retour

Objets membres

Le type de retour de std::not_fn contient un objet membre de type std::decay_t<F>.

Constructeurs

explicit T( F&& f );  // (1) (depuis C++17) (constexpr depuis C++20) ( exposition seulement* )

T( T&& f ) = default;
T( const T& f ) = default;  // (2) (depuis C++17)

  1. Le constructeur initialise directement sans liste (direct-non-list-initialization) l'objet membre (de type std::decay_t<F>) à partir de std::forward<F>(f). Lève toute exception levée par le constructeur sélectionné.

  2. Comme std::decay_t<F> doit satisfaire MoveConstructible, l'enveloppeur d'appel renvoyé est toujours MoveConstructible, et est CopyConstructible si std::decay_t<F> est CopyConstructible.

Les définitions explicitement par défaut rendent le type de retour non affectable.

(jusqu'à C++20)

Il n'est pas spécifié si ces constructeurs sont explicitement par défaut ni si le type de retour est affectable.

(depuis C++20)

  1. Équivalent à return !std::invoke(fd, std::forward<Args>(args)...);

  2. Équivalent à return !std::invoke(std::move(fd), std::forward<Args>(args)...);

  1. Équivalent en expression à !std::invoke(fd, std::forward<Args>(args)...)

  2. Équivalent en expression à !std::invoke(std::move(fd), std::forward<Args>(args)...)

(depuis C++20)

  1. Une valeur du type suivant.

std::not_fn type de retour sans état

Le type de retour est une classe sans état CopyConstructible. Il n'est pas spécifié si le type de retour est affectable.

  1. Ne lève aucune exception, sauf si la construction de fd lève une exception.

Implémentation possible

(1) not_fn

namespace detail
{
    template<class V, class F, class... Args>
    constexpr bool negate_invocable_impl = false;
    template<class F, class... Args>
    constexpr bool negate_invocable_impl<std::void_t<decltype(
        !std::invoke(std::declval<F>(), std::declval<Args>()...))>, F, Args...> = true;

    template<class F, class... Args>
    constexpr bool negate_invocable_v = negate_invocable_impl<void, F, Args...>;

    template<class F>
    struct not_fn_t
    {
        F f;

        template<class... Args,
            std::enable_if_t<negate_invocable_v<F&, Args...>, int> = 0>
        constexpr decltype(auto) operator()(Args&&... args) &
            noexcept(noexcept(!std::invoke(f, std::forward<Args>(args)...)))
        {
            return !std::invoke(f, std::forward<Args>(args)...);
        }

        template<class... Args,
            std::enable_if_t<negate_invocable_v<const F&, Args...>, int> = 0>
        constexpr decltype(auto) operator()(Args&&... args) const&
            noexcept(noexcept(!std::invoke(f, std::forward<Args>(args)...)))
        {
            return !std::invoke(f, std::forward<Args>(args)...);
        }

        template<class... Args,
            std::enable_if_t<negate_invocable_v<F, Args...>, int> = 0>
        constexpr decltype(auto) operator()(Args&&... args) &&
            noexcept(noexcept(!std::invoke(std::move(f), std::forward<Args>(args)...)))
        {
            return !std::invoke(std::move(f), std::forward<Args>(args)...);
        }

        template<class... Args,
            std::enable_if_t<negate_invocable_v<const F, Args...>, int> = 0>
        constexpr decltype(auto) operator()(Args&&... args) const&&
            noexcept(noexcept(!std::invoke(std::move(f), std::forward<Args>(args)...)))
        {
            return !std::invoke(std::move(f), std::forward<Args>(args)...);
        }

        // Les surcharges supprimées sont nécessaires depuis C++20
        // pour empêcher la sélection d'une surcharge non équivalente mais bien formée.

        template<class... Args,
            std::enable_if_t<!negate_invocable_v<F&, Args...>, int> = 0>
        void operator()(Args&&...) & = delete;

        template<class... Args,
            std::enable_if_t<!negate_invocable_v<const F&, Args...>, int> = 0>
        void operator()(Args&&...) const& = delete;

        template<class... Args,
            std::enable_if_t<!negate_invocable_v<F, Args...>, int> = 0>
        void operator()(Args&&...) && = delete;

        template<class... Args,
            std::enable_if_t<!negate_invocable_v<const F, Args...>, int> = 0>
        void operator()(Args&&...) const&& = delete;
    };
}

template<class F>
constexpr detail::not_fn_t<std::decay_t<F>> not_fn(F&& f)
{
    return {std::forward<F>(f)};
}

(2) not_fn

namespace detail
{
    template<auto ConstFn>
    struct stateless_not_fn
    {
        template<class... Args>
        constexpr auto operator()(Args&&... args) const
            noexcept(noexcept(!std::invoke(ConstFn, std::forward<Args>(args)...)))
            -> decltype(!std::invoke(ConstFn, std::forward<Args>(args)...))
        {
            return !std::invoke(ConstFn, std::forward<Args>(args)...);
        }
    };
}

template<auto ConstFn>
constexpr detail::stateless_not_fn<ConstFn> not_fn() noexcept
{
    if constexpr (std::is_pointer_v<decltype(ConstFn)> ||
                  std::is_member_pointer_v<decltype(ConstFn)>)
        static_assert(ConstFn != nullptr);

    return {};
}

Notes

std::not_fn est destiné à remplacer les négateurs std::not1 et std::not2 de l'ère C++03.

Macro de test de fonctionnalité | Valeur | Std | Fonctionnalité

__cpp_lib_not_fn | 201603L | (C++17) | std::not_fn(), (1)

202306L | (C++26) | Autoriser le passage d'objets appelables comme arguments de modèle non-type à std::not_fn, (2)

Exemple

#include <cassert>
#include <functional>

bool is_same(int a, int b) noexcept
{
    return a == b;
}

struct S
{
    int val;
    bool is_same(int arg) const noexcept { return val == arg; }
};

int main()
{
    // Utilisation avec une fonction libre :
    auto is_differ = std::not_fn(is_same);
    assert(is_differ(8, 8) == false); // équivalent à : !is_same(8, 8) == false
    assert(is_differ(6, 9) == true); // équivalent à : !is_same(8, 0) == true

    // Utilisation avec une fonction membre :
    auto member_differ = std::not_fn(&S::is_same);
    assert(member_differ(S{3}, 3) == false); //: S tmp{6}; !tmp.is_same(6) == false

    // La spécification noexcept est préservée :
    static_assert(noexcept(is_differ) == noexcept(is_same));
    static_assert(noexcept(member_differ) == noexcept(&S::is_same));

    // Utilisation avec un objet fonction :
    auto same = [](int a, int b) { return a == b; };
    auto differ = std::not_fn(same);
    assert(differ(1, 2) == true); //: !same(1, 2) == true
    assert(differ(2, 2) == false); //: !same(2, 2) == false

#if __cpp_lib_not_fn >= 202306L
    auto is_differ_cpp26 = std::not_fn<is_same>();
    assert(is_differ_cpp26(8, 8) == false);
    assert(is_differ_cpp26(6, 9) == true);

    auto member_differ_cpp26 = std::not_fn<&S::is_same>();
    assert(member_differ_cpp26(S{3}, 3) == false);

    auto differ_cpp26 = std::not_fn<same>();
    static_assert(differ_cpp26(1, 2) == true);
    static_assert(differ_cpp26(2, 2) == false);
#endif
}
