std::jmp_buf

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

Défini dans l'en-tête <csetjmp>

typedef /* unspecified */ jmp_buf;

Le type std::jmp_buf est un type tableau adapté au stockage des informations permettant de restaurer un environnement d'appel. Les informations stockées suffisent pour reprendre l'exécution au bon bloc du programme et à l'invocation de ce bloc. L'état des indicateurs d'état à virgule flottante, des fichiers ouverts ou de toute autre donnée n'est pas stocké dans un objet de type std::jmp_buf.

Exemple

#include <array>
#include <cmath>
#include <csetjmp>
#include <cstdlib>
#include <format>
#include <iostream>

std::jmp_buf solver_error_handler;

std::array<double, 2> solve_quadratic_equation(double a, double b, double c)
{
    const double discriminant = b * b - 4.0 * a * c;
    if (discriminant < 0)
        std::longjmp(solver_error_handler, true); // Aller au gestionnaire d'erreurs

    const double delta = std::sqrt(discriminant) / (2.0 * a);
    const double argmin = -b / (2.0 * a);
    return {argmin - delta, argmin + delta};
}

void show_quadratic_equation_solution(double a, double b, double c)
{
    std::cout << std::format("Solving {}x² + {}x + {} = 0...\n", a, b, c);
    auto [x_0, x_1] = solve_quadratic_equation(a, b, c);
    std::cout << std::format("x₁ = {}, x₂ = {}\n\n", x_0, x_1);
}

int main()
{
    if (setjmp(solver_error_handler))
    {
        // Gestionnaire d'erreurs pour le solveur
        std::cout << "No real solution\n";
        return EXIT_FAILURE;
    }

    for (auto [a, b, c] : {std::array{1, -3, 2}, {2, -3, -2}, {1, 2, 3}})
        show_quadratic_equation_solution(a, b, c);

    return EXIT_SUCCESS;
}

Output:

Solving 1x² + -3x + 2 = 0...
x₁ = 1, x₂ = 2

Solving 2x² + -3x + -2 = 0...
x₁ = -0.5, x₂ = 2

Solving 1x² + 2x + 3 = 0...
No real solution
