std::ranges::uninitialized_value_construct

Signature d'appel

template< no-throw-forward-iterator I, no-throw-sentinel-for<I> S >
    requires std::default_initializable<std::iter_value_t<I>>
I uninitialized_value_construct( I first, S last );  // (1) (depuis C++20) (constexpr depuis C++26)

template< no-throw-forward-range R >
    requires std::default_initializable<ranges::range_value_t<R>>
ranges::borrowed_iterator_t<R>
    uninitialized_value_construct( R&& r );  // (2) (depuis C++20) (constexpr depuis C++26)
  1. Construit des objets de type std::iter_value_t<I> dans la zone de mémoire non initialisée [first,last) par initialisation par valeur, comme si cela était fait par

for (; first != last; ++first) ::new (voidify(*first)) std::remove_reference_t<std::iter_reference_t<I>>(); return first;

Si une exception est levée pendant l'initialisation, les objets déjà construits sont détruits dans un ordre non spécifié.

  1. Équivalent à ranges::uninitialized_value_construct(ranges::begin(r), ranges::end(r)).

Les entités de type fonction décrites sur cette page sont des objets fonctions d'algorithme (appelés de manière informelle niebloids), c'est-à-dire :

  • Les listes d'arguments de modèle explicites ne peuvent pas être spécifiées lors de l'appel de l'un d'entre eux.
  • Aucun d'entre eux n'est visible par la recherche dépendante des arguments.
  • Lorsque l'un d'entre eux est trouvé par la recherche non qualifiée normale comme nom situé à gauche de l'opérateur d'appel de fonction, la recherche dépendante des arguments est inhibée.

Paramètres

  • first, last — paire itérateur-sentinelle désignant l'intervalle des éléments à initialiser par valeur
  • r — l'intervalle des éléments à initialiser par valeur

Valeur de retour

Comme décrit ci-dessus.

Complexité

Linéaire en la distance entre first et last.

Exceptions

Toute exception levée lors de la construction des éléments de l'intervalle de destination.

Notes

Une implémentation peut améliorer l'efficacité de ranges::uninitialized_value_construct, par exemple en utilisant ranges::fill, si le type de valeur de l'intervalle est un CopyAssignable TrivialType.

Implémentation possible

struct uninitialized_value_construct_fn
{
    template<no-throw-forward-iterator I, no-throw-sentinel-for<I> S>
        requires std::value_initializable<std::iter_value_t<I>>
    constexpr I operator()(I first, S last) const
    {
        using ValueType = std::remove_reference_t<std::iter_reference_t<I>>;
        if constexpr (std::is_trivially_default_constructible_v<ValueType>)
            return ranges::fill(first, last, ValueType());
        I rollback{first};
        try
        {
            for (; !(first == last); ++first)
                ::new (static_cast<void*>(std::addressof(*first))) ValueType();
            return first;
        }
        catch (...) // retour arrière : détruire les éléments construits
        {
            for (; rollback != first; ++rollback)
                ranges::destroy_at(std::addressof(*rollback));
            throw;
        }
    }

    template<no-throw-forward-range R>
        requires std::default_initializable<ranges::range_value_t<R>>
    constexpr ranges::borrowed_iterator_t<R> operator()(R&& r) const
    {
        return (*this)(ranges::begin(r), ranges::end(r));
    }
};

inline constexpr uninitialized_value_construct_fn uninitialized_value_construct{};

Exemple

#include <iostream>
#include <memory>
#include <string>

int main()
{
    struct S { std::string m{"▄▀▄▀▄▀▄▀"}; };

    constexpr int n{4};
    alignas(alignof(S)) char out[n * sizeof(S)];

    try
    {
        auto first{reinterpret_cast<S*>(out)};
        auto last{first + n};

        std::ranges::uninitialized_value_construct(first, last);

        auto count{1};
        for (auto it{first}; it != last; ++it)
            std::cout << count++ << ' ' << it->m << '\n';

        std::ranges::destroy(first, last);
    }
    catch (...)
    {
        std::cout << "Exception!\n";
    }

    // Pour les types scalaires, uninitialized_value_construct
    // remplit de zéros la zone de mémoire non initialisée donnée.
    int v[]{0, 1, 2, 3};
    std::cout << ' ';
    for (const int i : v)
        std::cout << ' ' << static_cast<char>(i + 'A');
    std::cout << "\n ";
    std::ranges::uninitialized_value_construct(std::begin(v), std::end(v));
    for (const int i : v)
        std::cout << ' ' << static_cast<char>(i + 'A');
    std::cout << '\n';
}
