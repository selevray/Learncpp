Compare les valeurs de pointeurs de deux unique_ptr, ou d'un unique_ptr et d'un nullptr.

1-7) Compare deux unique_ptr.

8-20) Compare un unique_ptr et un nullptr.

L'opérateur != est synthétisé à partir de operator==.

(depuis C++20)

Paramètres

  • x, y — unique_ptr à comparer

Valeur de retour

  1. x.get() == y.get()

  2. x.get() != y.get()

  3. std::less<CT>()(x.get(), y.get()), où CT est std::common_type<unique_ptr<T1, D1>::pointer, unique_ptr<T2, D2>::pointer>::type.

  4. !(y < x)

  5. y < x

  6. !(x < y)

  7. std::compare_three_way{}(x.get(), y.get())

8,9) !x

10,11) (bool)x

  1. std::less<unique_ptr<T,D>::pointer>()(x.get(), nullptr)

  2. std::less<unique_ptr<T,D>::pointer>()(nullptr, y.get())

  3. !(nullptr < x)

  4. !(y < nullptr)

  5. nullptr < x

  6. y < nullptr

  7. !(x < nullptr)

  8. !(nullptr < y)

  9. std::compare_three_way{}(x.get(), static_cast<typename unique_ptr<T, D>::pointer>(nullptr))

Exemple

#include <iostream>
#include <memory>

int main()
{
    std::unique_ptr<int> p1(new int(42));
    std::unique_ptr<int> p2(new int(42));

    std::cout << std::boolalpha
        << "(p1 == p1)       : " << (p1 == p1) << '\n'
        << "(p1 <=> p1) == 0 : " << ((p1 <=> p1) == 0) << '\n' // Depuis C++20

    // p1 et p2 pointent vers des emplacements mémoire différents, donc p1 != p2
        << "(p1 == p2)       : " << (p1 == p2) << '\n'
        << "(p1 < p2)        : " << (p1 < p2) << '\n'
        << "(p1 <=> p2) < 0  : " << ((p1 <=> p2) < 0) << '\n'   // Depuis C++20
        << "(p1 <=> p2) == 0 : " << ((p1 <=> p2) == 0) << '\n'; // Depuis C++20
}

Sortie possible :

(p1 == p1)       : true
(p1 <=> p1) == 0 : true
(p1 == p2)       : false
(p1 < p2)        : true
(p1 <=> p2) < 0  : true
(p1 <=> p2) == 0 : false
