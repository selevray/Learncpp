Construit un objet de type T et l'enveloppe dans un std::unique_ptr.

  1. Construit un type non tableau T. Les arguments args sont passés au constructeur de T. Cette surcharge ne participe à la résolution de surcharge que si T n'est pas un type tableau. La fonction est équivalente à :
unique_ptr<T>(new T(std::forward<Args>(args)...))
  1. Construit un tableau de taille dynamique donnée. Les éléments du tableau sont initialisés par valeur. Cette surcharge ne participe à la résolution de surcharge que si T est un tableau à borne inconnue. La fonction est équivalente à :
unique_ptr<T>(new std::remove_extent_t<T>[size]())

3,6) La construction de tableaux à borne connue est interdite.

  1. Identique à (1), sauf que l'objet est initialisé par défaut. Cette surcharge ne participe à la résolution de surcharge que si T n'est pas un type tableau. La fonction est équivalente à :
unique_ptr<T>(new T)
  1. Identique à (2), sauf que le tableau est initialisé par défaut. Cette surcharge ne participe à la résolution de surcharge que si T est un tableau à borne inconnue. La fonction est équivalente à :
unique_ptr<T>(new std::remove_extent_t<T>[size])

Paramètres

  • args — liste d'arguments avec lesquels une instance de T sera construite
  • size — la longueur du tableau à construire

Valeur de retour

std::unique_ptr d'une instance du type T.

Exceptions

Peut lever std::bad_alloc ou toute exception levée par le constructeur de T. Si une exception est levée, cette fonction n'a aucun effet.

Implémentation possible

make_unique (1-3)

// C++14 make_unique
namespace detail
{
    template<class>
    constexpr bool is_unbounded_array_v = false;
    template<class T>
    constexpr bool is_unbounded_array_v<T[]> = true;

    template<class>
    constexpr bool is_bounded_array_v = false;
    template<class T, std::size_t N>
    constexpr bool is_bounded_array_v<T[N]> = true;
} // espace de noms detail

template<class T, class... Args>
std::enable_if_t<!std::is_array<T>::value, std::unique_ptr<T>>
make_unique(Args&&... args)
{
    return std::unique_ptr<T>(new T(std::forward<Args>(args)...));
}

template<class T>
std::enable_if_t<detail::is_unbounded_array_v<T>, std::unique_ptr<T>>
make_unique(std::size_t n)
{
    return std::unique_ptr<T>(new std::remove_extent_t<T>[n]());
}

template<class T, class... Args>
std::enable_if_t<detail::is_bounded_array_v<T>> make_unique(Args&&...) = delete;

make_unique_for_overwrite (4-6)

// C++20 make_unique_for_overwrite
template<class T>
    requires (!std::is_array_v<T>)
std::unique_ptr<T> make_unique_for_overwrite()
{
    return std::unique_ptr<T>(new T);
}

template<class T>
    requires std::is_unbounded_array_v<T>
std::unique_ptr<T> make_unique_for_overwrite(std::size_t n)
{
    return std::unique_ptr<T>(new std::remove_extent_t<T>[n]);
}

template<class T, class... Args>
    requires std::is_bounded_array_v<T>
void make_unique_for_overwrite(Args&&...) = delete;

Notes

Contrairement à std::make_shared (qui possède std::allocate_shared), std::make_unique n'a pas d'équivalent prenant en charge les allocateurs. allocate_unique proposé dans P0211 nécessiterait d'inventer le type de suppresseur D pour le std::unique_ptr<T,D> qu'elle renvoie, qui contiendrait un objet allocateur et appellerait à la fois destroy et deallocate dans son operator().

#include <cstddef>
#include <iomanip>
#include <iostream>
#include <memory>
#include <utility>

struct Vec3
{
    int x, y, z;

    // Le constructeur suivant n'est plus nécessaire depuis C++20.
    Vec3(int x = 0, int y = 0, int z = 0) noexcept : x(x), y(y), z(z) {}

    friend std::ostream& operator<<(std::ostream& os, const Vec3& v)
    {
        return os << "{ x=" << v.x << ", y=" << v.y << ", z=" << v.z << " }";
    }
};

// Écrit les nombres de Fibonacci dans un itérateur de sortie.
template<typename OutputIt>
OutputIt fibonacci(OutputIt first, OutputIt last)
{
    for (int a = 0, b = 1; first != last; ++first)
    {
        *first = b;
        b += std::exchange(a, b);
    }
    return first;
}

int main()
{
    // Utilise le constructeur par défaut.
    std::unique_ptr<Vec3> v1 = std::make_unique<Vec3>();
    // Utilise le constructeur correspondant à ces arguments.
    std::unique_ptr<Vec3> v2 = std::make_unique<Vec3>(0, 1, 2);
    // Crée un unique_ptr vers un tableau de 5 éléments.
    std::unique_ptr<Vec3[]> v3 = std::make_unique<Vec3[]>(5);

    // Crée un unique_ptr vers un tableau non initialisé de 10 entiers,
    // puis le remplit avec les nombres de Fibonacci.
    std::unique_ptr<int[]> i1 = std::make_unique_for_overwrite<int[]>(10);
    fibonacci(i1.get(), i1.get() + 10);

    std::cout << "make_unique<Vec3>():      " << *v1 << '\n'
              << "make_unique<Vec3>(0,1,2): " << *v2 << '\n'
              << "make_unique<Vec3[]>(5):   ";
    for (std::size_t i = 0; i < 5; ++i)
        std::cout << std::setw(i ? 30 : 0) << v3[i] << '\n';
    std::cout << '\n';

    std::cout << "make_unique_for_overwrite<int[]>(10), fibonacci(...): [" << i1[0];
    for (std::size_t i = 1; i < 10; ++i)
        std::cout << ", " << i1[i];
    std::cout << "]\n";
}
