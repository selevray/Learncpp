std::regex_traits<CharT>::value

Traduction automatique (Claude Haiku 5.5) de cppreference.com, sous licence CC-BY-SA 3.0.

int value( CharT ch, int radix ) const;  // (depuis C++11)

Détermine la valeur représentée par le chiffre ch dans la base numérique radix, selon la locale actuellement imprégnée. Cette fonction est appelée par std::regex lors du traitement de quantificateurs tels que {1} ou {2,5}, de références arrières telles que \1, et des échappements de caractères hexadécimaux et Unicode.

Paramètres

  • ch — le caractère pouvant représenter un chiffre
  • radix — 8, 10 ou 16

Valeur de retour

La valeur numérique si ch représente effectivement un chiffre de la locale actuellement imprégnée valide pour la base numérique radix, ou -1 en cas d'erreur.

Exemple

#include <iostream>
#include <locale>
#include <map>
#include <regex>

// Ce trait de regex personnalisé permet les chiffres japonais
struct jnum_traits : std::regex_traits<wchar_t>
{
    static std::map<wchar_t, int> data;
    int value(wchar_t ch, int radix) const
    {
        wchar_t up = std::toupper(ch, getloc());
        return data.count(up) ? data[up] : regex_traits::value(ch, radix);
    }
};
std::map<wchar_t, int> jnum_traits::data = {{L'〇',0}, {L'一',1}, {L'二',2},
                                            {L'三',3}, {L'四',4}, {L'五',5},
                                            {L'六',6}, {L'七',7}, {L'八',8},
                                            {L'九',9}, {L'Ａ',10}, {L'Ｂ',11},
                                            {L'Ｃ',12}, {L'Ｄ',13}, {L'Ｅ',14},
                                            {L'Ｆ',15}};

int main()
{
    std::locale::global(std::locale("ja_JP.utf8"));
    std::wcout.sync_with_stdio(false);
    std::wcout.imbue(std::locale());

    std::wstring in = L"風";

    if (std::regex_match(in, std::wregex(L"\\u98a8")))
        std::wcout << "\\u98a8 matched " << in << '\n';

    if (std::regex_match(in, std::basic_regex<wchar_t, jnum_traits>(L"\\u九八ａ八")))
        std::wcout << L"\\u九八ａ八 with custom traits matched " << in << '\n';
}

Output:

\u98a8 matched 風
\u九八ａ八 with custom traits matched 風
